   530→  // Relative view distance helper for 'Recently Viewed' label
   531→  const getRecentlyViewedMsg = (folderId: string) => {
   532→    const timestamp = recentlyViewed[folderId];
   533→    if (!timestamp) return null;
   534→    const diff = Date.now() - timestamp;
   535→    if (diff < 60000) return 'Just now';
   536→    const mins = Math.floor(diff / 60000);
   537→    if (mins < 60) return `${mins}m ago`;
   538→    const hrs = Math.floor(diff / 3600000);
   539→    if (hrs < 24) return `${hrs}h ago`;
   540→    const days = Math.floor(hrs / 24);
   541→    return `${days}d ago`;
   542→  };
