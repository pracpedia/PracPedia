   137→   137→/* ---------------- AnimatedStat (kept — one-shot rAF count-up) ---------------- */
   138→   138→
   139→   139→const AnimatedStat: React.FC<{ value: number; className?: string }> = ({ value, className = '' }) => {
   140→   140→  const ref = useRef<HTMLSpanElement>(null);
   141→   141→  const inView = useInView(ref, { once: true, margin: '-40px' });
   142→   142→  const [display, setDisplay] = useState(0);
   143→   143→
   144→   144→  useEffect(() => {
   145→   145→    if (!inView) return;
   146→   146→    const prefersReducedMotion =
   147→   147→      typeof window !== 'undefined' && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
   148→   148→    if (prefersReducedMotion) {
   149→   149→      setDisplay(value);
   150→   150→      return;
   151→   151→    }
   152→   152→    let raf = 0;
   153→   153→    const start = performance.now();
   154→   154→    const duration = 1400;
   155→   155→    const tick = (now: number) => {
   156→   156→      const t = Math.min((now - start) / duration, 1);
   157→   157→      const eased = 1 - Math.pow(1 - t, 3);
   158→   158→      setDisplay(Math.round(eased * value));
   159→   159→      if (t < 1) raf = requestAnimationFrame(tick);
   160→   160→    };
   161→   161→    raf = requestAnimationFrame(tick);
   162→   162→    return () => cancelAnimationFrame(raf);
   163→   163→  }, [inView, value]);
   164→   164→
   165→   165→  return <span ref={ref} className={className}>{display}</span>;
   166→   166→};

   180→   180→  const [openFaq, setOpenFaq] = useState<number | null>(0);
   181→   181→
   182→   182→  /* GSAP refs for the 3D calculus surface animation */
   183→   183→  const sceneRef = useRef<HTMLDivElement>(null);
   184→   184→  const paraboloidRef = useRef<SVGGElement>(null);

   202→   202→
   203→   203→  /* GSAP-animated 3D squared grid surface — infinite circular rotation */
   204→   204→  useEffect(() => {
   205→   205→    if (typeof window === 'undefined') return;
   206→   206→    const prefersReduced =
   207→   207→      typeof window !== 'undefined' &&
   208→   208→      window.matchMedia('(prefers-reduced-motion: reduce)').matches;
   209→   209→
   210→   210→    const tweens: gsap.core.Tween[] = [];
   211→   211→    const timelines: gsap.core.Timeline[] = [];
   212→   212→
   213→   213→    // ─────────────────────────────────────────────────────────────────────────
   214→   214→    // Multivariable wavy function:  z = sin(x) · cos(y)
   215→   215→    // Classic egg-crate surface — waves in both x and y directions.
   216→   216→    // Rendered as a grid of SQUARED cells (quadrilaterals), not curved
   217→   217→    // wireframe lines. Each cell is colored by its height for a heat-map
   218→   218→    // effect. The entire grid rotates in circular motion around the z axis.
   219→   219→    // ─────────────────────────────────────────────────────────────────────────
   220→   220→    function calcSurface(x: number, y: number): number {
   221→   221→      return Math.sin(x) * Math.cos(y);
   222→   222→    }
   223→   223→
   224→   224→    // ── Grid configuration ──
   225→   225→    const GRID_N = 10;        // 10×10 = 100 cells
   226→   226→    const GRID_RANGE = 2.5;   // x,y from -2.5 to +2.5
   227→   227→    const CELL_SIZE = (2 * GRID_RANGE) / GRID_N;
   228→   228→
   229→   229→    // Pre-compute grid corner points in math coords (static — never changes)
   230→   230→    // gridPoints[i][j] = { x, y, z } for the corner at row i, col j
   231→   231→    const gridPoints: Array<Array<{ x: number; y: number; z: number }>> = [];
   232→   232→    for (let i = 0; i <= GRID_N; i++) {
   233→   233→      const row: Array<{ x: number; y: number; z: number }> = [];
   234→   234→      for (let j = 0; j <= GRID_N; j++) {
   235→   235→        const x = -GRID_RANGE + j * CELL_SIZE;
   236→   236→        const y = -GRID_RANGE + i * CELL_SIZE;
   237→   237→        row.push({ x, y, z: calcSurface(x, y) });
   238→   238→      }
   239→   239→      gridPoints.push(row);
   240→   240→    }
   241→   241→
   242→   242→    // 5 evaluation points
   243→   243→    const evalPoints: Array<{ x: number; y: number; label: string }> = [
   244→   244→      { x: 0, y: 0, label: 'f(0, 0)' },
   245→   245→      { x: 1.5, y: 1.0, label: 'f(1.5, 1.0)' },
   246→   246→      { x: -1.2, y: 0.8, label: 'f(-1.2, 0.8)' },
   247→   247→      { x: 2.0, y: -1.5, label: 'f(2.0, -1.5)' },
   248→   248→      { x: -2.0, y: -1.0, label: 'f(-2.0, -1.0)' },
   249→   249→    ];
   250→   250→
   251→   251→    // ── 3D projection with Z-axis rotation + X-axis tilt oscillation ──
   252→   252→    const SVG_W = 240;
   253→   253→    const SVG_H = 160;
   254→   254→    const CX = SVG_W / 2;
   255→   255→    const CY = SVG_H / 2 + 8;
   256→   256→    const SCALE = 26;
   257→   257→    const TILT_BASE = Math.PI / 6;
   258→   258→    const Z_SCALE = 28;
   259→   259→
   260→   260→    const rotation = { angle: 0, tilt: 0 };
   261→   261→
   262→   262→    function project(x: number, y: number, z: number, rotZ: number, tilt: number) {
   263→   263→      const cosR = Math.cos(rotZ);
   264→   264→      const sinR = Math.sin(rotZ);
   265→   265→      const xR = x * cosR - y * sinR;
   266→   266→      const yR = x * sinR + y * cosR;
   267→   267→      const totalTilt = TILT_BASE + tilt;
   268→   268→      const zPx = z * Z_SCALE;
   269→   269→      const yTilted = yR * Math.cos(totalTilt) - zPx * Math.sin(totalTilt);
   270→   270→      const zPrime = yR * Math.sin(totalTilt) + zPx * Math.cos(totalTilt);
   271→   271→      return { x: CX + xR * SCALE, y: CY - yTilted * SCALE, depth: zPrime };
   272→   272→    }
   273→   273→
   274→   274→    // Helper: interpolate color from z value
   275→   275→    // z ranges roughly -1..+1 (since sin·cos ranges -1..+1)
   276→   276→    // Low (negative z) → deep cyan/teal, mid (0) → slate, high (positive z) → indigo/violet
   277→   277→    function zToColor(z: number): string {
   278→   278→      const t = (z + 1) / 2; // normalize to 0..1
   279→   279→      if (t < 0.25) return `rgba(8, 145, 178, ${(0.15 + t * 0.4).toFixed(2)})`;  // deep cyan
   280→   280→      if (t < 0.5)  return `rgba(34, 211, 238, ${(0.2 + t * 0.4).toFixed(2)})`;  // cyan
   281→   281→      if (t < 0.75) return `rgba(129, 140, 248, ${(0.2 + t * 0.4).toFixed(2)})`; // indigo
   282→   282→      return `rgba(167, 139, 250, ${(0.25 + t * 0.35).toFixed(2)})`;              // violet
   283→   283→    }
   284→   284→
   285→   285→    // ── Re-project everything at the current rotation ──
   286→   286→    let currentEvalIdx = 0;
   287→   287→    function renderAtRotation(rotZ: number, tilt: number, evalIdx: number) {
   288→   288→      // Squared grid cells — update each polygon's points + fill
   289→   289→      const cellEls = paraboloidRef.current?.querySelectorAll('polygon');
   290→   290→      if (cellEls) {
   291→   291→        let cellIdx = 0;
   292→   292→        for (let i = 0; i < GRID_N; i++) {
   293→   293→          for (let j = 0; j < GRID_N; j++) {
   294→   294→            const el = cellEls[cellIdx] as SVGPolygonElement | undefined;
   295→   295→            cellIdx++;
   296→   296→            if (!el) continue;
   297→   297→            // 4 corners of this cell
   298→   298→            const c00 = project(gridPoints[i][j].x, gridPoints[i][j].y, gridPoints[i][j].z, rotZ, tilt);
   299→   299→            const c01 = project(gridPoints[i][j + 1].x, gridPoints[i][j + 1].y, gridPoints[i][j + 1].z, rotZ, tilt);
   300→   300→            const c11 = project(gridPoints[i + 1][j + 1].x, gridPoints[i + 1][j + 1].y, gridPoints[i + 1][j + 1].z, rotZ, tilt);
   301→   301→            const c10 = project(gridPoints[i + 1][j].x, gridPoints[i + 1][j].y, gridPoints[i + 1][j].z, rotZ, tilt);
   302→   302→            el.setAttribute('points',
   303→   303→              `${c00.x.toFixed(1)},${c00.y.toFixed(1)} ` +
   304→   304→              `${c01.x.toFixed(1)},${c01.y.toFixed(1)} ` +
   305→   305→              `${c11.x.toFixed(1)},${c11.y.toFixed(1)} ` +
   306→   306→              `${c10.x.toFixed(1)},${c10.y.toFixed(1)}`
   307→   307→            );
   308→   308→            // Color by average z
   309→   309→            const avgZ = (gridPoints[i][j].z + gridPoints[i][j + 1].z + gridPoints[i + 1][j + 1].z + gridPoints[i + 1][j].z) / 4;
   310→   310→            el.setAttribute('fill', zToColor(avgZ));
   311→   311→            // Depth-based opacity
   312→   312→            const avgDepth = (c00.depth + c01.depth + c11.depth + c10.depth) / 4;
   313→   313→            const depthOpacity = 0.4 + 0.5 * (1 - Math.max(0, Math.min(1, (avgDepth + 40) / 80)));
   314→   314→            el.setAttribute('opacity', depthOpacity.toFixed(2));
   315→   315→          }
   316→   316→        }
   317→   317→      }
   318→   318→
   319→   319→      // Grid lines — update each line's endpoints
   320→   320→      const lineEls = paraboloidRef.current?.querySelectorAll('line.grid-line');
   321→   321→      if (lineEls) {
   322→   322→        let lineIdx = 0;
   323→   323→        // Horizontal lines (constant y, varying x)
   324→   324→        for (let i = 0; i <= GRID_N; i++) {
   325→   325→          const start = project(gridPoints[i][0].x, gridPoints[i][0].y, gridPoints[i][0].z, rotZ, tilt);
   326→   326→          const end = project(gridPoints[i][GRID_N].x, gridPoints[i][GRID_N].y, gridPoints[i][GRID_N].z, rotZ, tilt);
   327→   327→          if (lineEls[lineIdx]) {

   329→   329→            (lineEls[lineIdx] as SVGLineElement).setAttribute('y1', String(start.y.toFixed(1)));
   330→   330→            (lineEls[lineIdx] as SVGLineElement).setAttribute('x2', String(end.x.toFixed(1)));
   331→   331→            (lineEls[lineIdx] as SVGLineElement).setAttribute('y2', String(end.y.toFixed(1)));
   332→   332→          }
   333→   333→          lineIdx++;
   334→   334→        }
   335→   335→        // Vertical lines (constant x, varying y)
   336→   336→        for (let j = 0; j <= GRID_N; j++) {
   337→   337→          const start = project(gridPoints[0][j].x, gridPoints[0][j].y, gridPoints[0][j].z, rotZ, tilt);
   338→   338→          const end = project(gridPoints[GRID_N][j].x, gridPoints[GRID_N][j].y, gridPoints[GRID_N][j].z, rotZ, tilt);
   339→   339→          if (lineEls[lineIdx]) {
   340→   340→            (lineEls[lineIdx] as SVGLineElement).setAttribute('x1', String(start.x.toFixed(1)));
   341→   341→            (lineEls[lineIdx] as SVGLineElement).setAttribute('y1', String(start.y.toFixed(1)));
   342→   342→            (lineEls[lineIdx] as SVGLineElement).setAttribute('x2', String(end.x.toFixed(1)));
   343→   343→            (lineEls[lineIdx] as SVGLineElement).setAttribute('y2', String(end.y.toFixed(1)));
   344→   344→          }
   345→   345→          lineIdx++;
   346→   346→        }
   347→   347→      }
   348→   348→
   349→   349→      // 3D coordinate system axes (same as before)
   350→   350→      if (axesRef.current) {
   351→   351→        const origin = project(0, 0, 0, rotZ, tilt);
   352→   352→        const xEnd = project(2.5, 0, 0, rotZ, tilt);
   353→   353→        const yEnd = project(0, 2.5, 0, rotZ, tilt);
   354→   354→        const zEnd = project(0, 0, 0.8, rotZ, tilt);
   355→   355→        const axisLines = axesRef.current.querySelectorAll('.axis-line');
   356→   356→        const axisLabels = axesRef.current.querySelectorAll('.axis-label');
   357→   357→        const axisTicks = axesRef.current.querySelectorAll('.axis-tick');
   358→   358→        if (axisLines.length >= 3) {
   359→   359→          (axisLines[0] as SVGLineElement).setAttribute('x1', String(origin.x));
   360→   360→          (axisLines[0] as SVGLineElement).setAttribute('y1', String(origin.y));
   361→   361→          (axisLines[0] as SVGLineElement).setAttribute('x2', String(xEnd.x));
   362→   362→          (axisLines[0] as SVGLineElement).setAttribute('y2', String(xEnd.y));
   363→   363→          (axisLines[1] as SVGLineElement).setAttribute('x1', String(origin.x));
   364→   364→          (axisLines[1] as SVGLineElement).setAttribute('y1', String(origin.y));
   365→   365→          (axisLines[1] as SVGLineElement).setAttribute('x2', String(yEnd.x));
   366→   366→          (axisLines[1] as SVGLineElement).setAttribute('y2', String(yEnd.y));
   367→   367→          (axisLines[2] as SVGLineElement).setAttribute('x1', String(origin.x));
   368→   368→          (axisLines[2] as SVGLineElement).setAttribute('y1', String(origin.y));
   369→   369→          (axisLines[2] as SVGLineElement).setAttribute('x2', String(zEnd.x));
   370→   370→          (axisLines[2] as SVGLineElement).setAttribute('y2', String(zEnd.y));
   371→   371→        }
   372→   372→        if (axisTicks.length >= 6) {
   373→   373→          for (let i = 0; i < 2; i++) {
   374→   374→            const tickX = project(i + 1, 0, 0, rotZ, tilt);
   375→   375→            const tickY = project(0, i + 1, 0, rotZ, tilt);
