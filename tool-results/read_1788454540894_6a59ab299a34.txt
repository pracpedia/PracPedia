    35→interface SubjectType {
    36→  id: string;
    37→  title: string;
    38→  description: string;

    40→
    41→interface MessageType {
    42→  id?: string;
    43→  _id?: string;
    44→  userId: string;
    45→  userName: string;
    46→  userRole: 'user' | 'admin';
    47→  content: string;
    48→  subjectId: string;
    49→  createdAt: string;
    50→  userProfilePic?: string;
    51→  imageUrl?: string;

    54→interface ClassroomDiscussionProps {

    57→
    58→export const ClassroomDiscussion: React.FC<ClassroomDiscussionProps> = ({ subjects }) => {
    59→  const { user, apiFetch } = useAuth();
    60→  const { language, t } = useLanguage();
    61→  const [messages, setMessages] = useState<MessageType[]>([]);
    62→  const messagesRef = useRef<MessageType[]>([]);
    63→  const [activeChannelId, setActiveChannelId] = useState<string>('general');
    64→  const [typedMessage, setTypedMessage] = useState<string>('');
    65→  const [pendingImage, setPendingImage] = useState<string | null>(null);
    66→  const [zoomedImage, setZoomedImage] = useState<string | null>(null);
    67→  const [revealedTimeId, setRevealedTimeId] = useState<string | null>(null);
    68→  const imageInputRef = useRef<HTMLInputElement>(null);
    69→  const [loading, setLoading] = useState<boolean>(false);
    70→  const [submitting, setSubmitting] = useState<boolean>(false);
    71→  const [errorMessage, setErrorMessage] = useState<string | null>(null);
    72→
    73→  // Close zoom popup on Escape
    74→  useEffect(() => {
    75→    if (!zoomedImage) return;
    76→    const handler = (e: KeyboardEvent) => {
    77→      if (e.key === 'Escape') setZoomedImage(null);
    78→    };
    79→    window.addEventListener('keydown', handler);
    80→    return () => window.removeEventListener('keydown', handler);
    81→  }, [zoomedImage]);
    82→
    83→  // Custom mobile screen flow: 'channels' list vs the 'chat' window itself
    84→  const [mobileView, setMobileView] = useState<'channels' | 'chat'>('chat');
    85→
    86→  const scrollContainerRef = useRef<HTMLDivElement>(null);
    87→
    88→  // Keep messagesRef in sync with messages state so the long-poll loop can
    89→  // read the latest message timestamp without being a dependency of the effect.
    90→  useEffect(() => {
    91→    messagesRef.current = messages;
    92→  }, [messages]);
    93→
    94→  // Real-time active users
    95→  const [activeUsers, setActiveUsers] = useState<any[]>((window as any).__activeUsers || []);
    96→
    97→  // Quick Action triggers removed per user request — chat input is now clean.
    98→
    99→  // Fetch messages of specified subject channel.
   100→  // If `since` is provided, uses long-polling (server holds up to 20s waiting
   101→  // for new messages). Otherwise, fetches the latest 200 messages immediately.
   102→  const fetchChannelMessages = async (subjId: string, silent: boolean = false, since?: string) => {
   103→    if (!silent) {
   104→      setLoading(true);
   105→      setErrorMessage(null);
   106→    }
   107→    try {
   108→      // Always use /api/chat/[subjectId] — 'general' maps to global (null subjectId)
   109→      const endpoint = `/api/chat/${subjId}` + (since ? `?since=${encodeURIComponent(since)}` : '');
   110→      const res = await apiFetch(endpoint);
   111→      if (res.ok) {
   112→        const raw = await res.json();
   113→        // Response shape: { messages: [...], polledAt: '...' }
   114→        // Handle both old (array) and new ({messages}) shapes for resilience
   115→        const rawMessages = Array.isArray(raw) ? raw : (raw.messages || []);
   116→        // Normalize API fields (text -> content, userAvatar -> userProfilePic) to match local MessageType shape
   117→        const data: MessageType[] = rawMessages.map((m: any) => ({
   118→          id: m.id,
   119→          _id: m._id,
   120→          userId: m.userId,
   121→          userName: m.userName,
   122→          userRole: m.userRole === 'admin' ? 'admin' : 'user',
   123→          content: m.text ?? m.content ?? '',
   124→          subjectId: m.subjectId ?? subjId,
   125→          createdAt: m.createdAt,
   126→          userProfilePic: m.userProfilePic ?? m.userAvatar,
   127→          imageUrl: m.imageUrl,
   128→        }));
   129→        setMessages((prev) => {
   130→          if (since) {
   131→            // Long-poll mode — merge new messages into existing
   132→            if (data.length === 0) return prev;
   133→            const existingIds = new Set(prev.map(m => m.id || m._id));
   134→            const newOnes = data.filter(m => !existingIds.has(m.id || m._id));
   135→            return newOnes.length > 0 ? [...prev, ...newOnes] : prev;
   136→          }
   137→          // Initial fetch — replace all
   138→          return data;
   139→        });
   140→      } else if (!silent) {
   141→        const errData = await res.json();
   142→        setErrorMessage(errData.error || t('networkLoss'));
   143→      }
   144→    } catch (e: any) {
   145→      if (!silent) {
   146→        setErrorMessage(t('networkLoss'));
   147→      }
   148→    } finally {
   149→      if (!silent) {
   150→        setLoading(false);
   151→      }
   152→    }
   153→  };
   154→
   155→  // Image picker — converts to base64 data URL for inline upload
   156→  const handleImagePick = (e: React.ChangeEvent<HTMLInputElement>) => {
   157→    const file = e.target.files?.[0];
   158→    if (imageInputRef.current) imageInputRef.current.value = '';
   159→    if (!file) return;
   160→    if (!file.type.startsWith('image/')) {
   161→      setErrorMessage('Please choose an image file.');
   162→      return;
   163→    }
   164→    if (file.size > 2 * 1024 * 1024) {
   165→      setErrorMessage('Max 2 MB for chat images.');
   166→      return;
   167→    }
   168→    const reader = new FileReader();
   169→    reader.onload = () => setPendingImage(String(reader.result || ''));
   170→    reader.onerror = () => setErrorMessage('Could not read image file.');
   171→    reader.readAsDataURL(file);
   172→  };
   173→
   174→  // Perform post of typed message content + optional image
   175→  const handleSendMessage = async (incomingContent?: string) => {
   176→    const targetContent = incomingContent || typedMessage;
   177→    // Need either text or an image to send
   178→    if ((!targetContent.trim() && !pendingImage) || submitting) return;
   179→
   180→    setSubmitting(true);
   181→    setErrorMessage(null);
   182→
   183→    const messageContent = targetContent;
   184→    const imageToSend = pendingImage;
   185→    if (!incomingContent) {
   186→      setTypedMessage('');
   187→    }
   188→    setPendingImage(null);
   189→
   190→    try {
   191→      // Next.js /api/chat expects { text, subjectId, imageUrl } — pass null for general lounge
   192→      const subjectPayload = activeChannelId === 'general' ? null : activeChannelId;
   193→      const res = await apiFetch('/api/chat', {
   194→        method: 'POST',
   195→        headers: {
   196→          'Content-Type': 'application/json',
   197→        },
   198→        body: JSON.stringify({
   199→          text: messageContent || '(image)',
   200→          subjectId: subjectPayload,
   201→          imageUrl: imageToSend || undefined,
   202→        })
   203→      });
   204→
   205→      if (res.ok) {
   206→        const newMsg = await res.json();
   207→        // Normalize the new message to match local MessageType shape
   208→        const normalized: MessageType = {
   209→          id: newMsg.id,
   210→          _id: newMsg._id,
   211→          userId: newMsg.userId,
   212→          userName: newMsg.userName,
   213→          userRole: newMsg.userRole === 'admin' ? 'admin' : 'user',
   214→          content: newMsg.text ?? newMsg.content ?? messageContent,
   215→          subjectId: newMsg.subjectId ?? activeChannelId,
   216→          createdAt: newMsg.createdAt,
   217→          userProfilePic: newMsg.userProfilePic ?? newMsg.userAvatar,
   218→          imageUrl: newMsg.imageUrl,
   219→        };
   220→        setMessages((prev) => [...prev, normalized]);
   221→        // Force scroll-to-bottom on newly typed posts from our side
   222→        setTimeout(() => {
   223→          if (scrollContainerRef.current) {
   224→            scrollContainerRef.current.scrollTop = scrollContainerRef.current.scrollHeight;
   225→          }
   226→        }, 80);
   227→      } else {
   228→        const errData = await res.json();
   229→        setErrorMessage(errData.error || "Failed to broadcast message.");
   230→        if (!incomingContent) setTypedMessage(messageContent); // Restore text on failure
   231→      }
   232→    } catch (err) {
   233→      setErrorMessage("Broadcast failed due to structural network discrepancy.");
   234→      if (!incomingContent) setTypedMessage(messageContent); // Restore text on failure
   235→    } finally {
   236→      setSubmitting(false);
   237→    }
   238→  };
   239→
   240→  const handleFormSubmit = async (e: React.FormEvent) => {
   241→    e.preventDefault();
   242→    await handleSendMessage();
   243→  };
   244→
   245→  // Perform message eradication (admin or owner only)
   246→  const handleDeleteMessage = async (msgId: string) => {
   247→    if (!msgId) return;
   248→
   249→    try {
   250→      // Next.js /api/chat delete via ?id= query param
   251→      const res = await apiFetch(`/api/chat?id=${msgId}`, {
   252→        method: 'DELETE',
   253→      });
   254→
   255→      if (res.ok) {
   256→        // Remove locally from UI state
   257→        setMessages((prev) => prev.filter(m => (m.id || m._id) !== msgId));
   258→      } else {
   259→        const errData = await res.json();
   260→        setErrorMessage(errData.error || "Could not delete message."); setTimeout(() => setErrorMessage(null), 4000);
   261→      }
   262→    } catch (err) {
   263→      setErrorMessage("Network error deleting message."); setTimeout(() => setErrorMessage(null), 4000);
   264→    }
   265→  };
   266→
   267→  // ── Client-side polling (Vercel serverless-safe) ──────────────────────
   268→  // Instead of long-polling (server holds request open 20s — burns Vercel
   269→  // quota + times out on Hobby plan), the client polls every 3 seconds.
   270→  // The server returns immediately with any new messages since the last
   271→  // poll. This is cheaper, simpler, and works on all serverless platforms.
   272→  useEffect(() => {
   273→    let cancelled = false;
   274→
   275→    const startPollLoop = async (subjId: string) => {
   276→      // Initial fetch (no `since` param — get latest 200 messages)
   277→      await fetchChannelMessages(subjId, false);
   278→
   279→      if (cancelled) return;
   280→
   281→      // Poll loop — fetch new messages every 3 seconds
   282→      while (!cancelled) {
   283→        await new Promise((r) => setTimeout(r, 3000));
   284→        if (cancelled) break;
   285→
   286→        // Get the latest message timestamp to use as the `since` cursor
   287→        const latestMsg = messagesRef.current[messagesRef.current.length - 1];
   288→        const since = latestMsg?.createdAt || new Date(0).toISOString();
   289→
   290→        // Fetch only new messages since the last poll
   291→        await fetchChannelMessages(subjId, true, since);
   292→      }
   293→    };
   294→
   295→    startPollLoop(activeChannelId);
   296→
   297→    return () => {
   298→      cancelled = true;
   299→    };
   300→  }, [activeChannelId]);
   301→
   302→  // Listen to WebSocket-triggered immediate reload event for instantaneous responsiveness
   303→  useEffect(() => {
   304→    const handleRemoteChatUpdate = (e: Event) => {
   305→      const customEvent = e as CustomEvent;
   306→      if (customEvent.detail && customEvent.detail.subjectId === activeChannelId) {
   307→        fetchChannelMessages(activeChannelId, true);
   308→      }
   309→    };
   310→
   311→    window.addEventListener('chat-updated', handleRemoteChatUpdate);
   312→    return () => {
   313→      window.removeEventListener('chat-updated', handleRemoteChatUpdate);
   314→    };
   315→  }, [activeChannelId]);
   316→
   317→  // Sync initial components presence data and update when event triggers
   318→  useEffect(() => {
   319→    if ((window as any).__activeUsers) {
   320→      setActiveUsers((window as any).__activeUsers);
   321→    }
   322→    const handlePresenceUpdated = (e: Event) => {
   323→      const customEvent = e as CustomEvent;
   324→      if (customEvent.detail && customEvent.detail.activeUsers) {
   325→        setActiveUsers(customEvent.detail.activeUsers);
   326→      }
   327→    };
   328→    window.addEventListener('presence-updated', handlePresenceUpdated);
   329→    return () => {
   330→      window.removeEventListener('presence-updated', handlePresenceUpdated);
   331→    };
   332→  }, []);
   333→
   334→  // Handle user-friendly quiet auto-scrolling on load or when sending message
   335→  const lastMessageCountRef = useRef<number>(0);
   336→  useEffect(() => {
   337→    const scrollContainer = scrollContainerRef.current;
   338→    if (!scrollContainer) return;
   339→
   340→    if (messages.length > lastMessageCountRef.current) {
   341→      const lastMsg = messages[messages.length - 1];
   342→      const isMyMessage = lastMsg && lastMsg.userId === user?.id;
   343→      const isInitialScroll = !loading && lastMessageCountRef.current === 0;
   344→
   345→      // Scroll to bottom immediately if it's the initial load or if the user sent a new message
   346→      if (isInitialScroll || isMyMessage) {
   347→        scrollContainer.scrollTop = scrollContainer.scrollHeight;
   348→      } else {
   349→        // If someone else posted, scroll down only if the user is already near the bottom
   350→        // (to prevent interrupting their reading of history)
   351→        const threshold = 180; // pixels from bottom threshold zone
   352→        const isNearBottom = scrollContainer.scrollHeight - scrollContainer.scrollTop - scrollContainer.clientHeight < threshold;
   353→        if (isNearBottom) {
   354→          scrollContainer.scrollTop = scrollContainer.scrollHeight;
   355→        }
   356→      }
   357→    }
   358→    lastMessageCountRef.current = messages.length;
   359→  }, [messages, loading, user?.id]);
   360→
   361→  const activeChannelObj = subjects.find(s => s.id === activeChannelId);
   362→  const activeChannelTitle = activeChannelId === 'general' ? 'General Academic Lounge' : `${activeChannelObj?.title} Support Channel`;
   363→  const activeChannelDesc = activeChannelId === 'general'
   364→    ? 'Universal classroom bulletin board and loose peer conversation space.'
   365→    : activeChannelObj?.description;
   366→
   367→  const formatDateLabel = (isoStr: string) => {
   368→    try {
   369→      const date = new Date(isoStr);
   370→      return date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
   371→    } catch (e) {
   372→      return '';
   373→    }
   374→  };
   375→
   376→  // Assign specific visual descriptors, emojis, and icons of subjects for professional feel
   377→  const getSubjectIcon = (title: string, sizeClass = "w-4 h-4") => {
   378→    const l = title.toLowerCase();
   379→    if (l.includes('physic')) return <Compass className={`${sizeClass} text-indigo-400`} />;
   380→    if (l.includes('chemist')) return <Beaker className={`${sizeClass} text-emerald-400`} />;
   381→    if (l.includes('biolog')) return <Sparkle className={`${sizeClass} text-amber-400`} />;
   382→    if (l.includes('math')) return <GraduationCap className={`${sizeClass} text-pink-400`} />;
   383→    if (l.includes('ict')) return <Code className={`${sizeClass} text-cyan-400`} />;
   384→    return <Hash className={`${sizeClass} text-slate-450`} />;
   385→  };
   386→
   387→  const getSubjectGradient = (title: string) => {
   388→    const l = title.toLowerCase();
   389→    if (l.includes('physic')) return 'from-indigo-505/20 to-indigo-950/25 border-indigo-500/20 text-indigo-350';
   390→    if (l.includes('chemist')) return 'from-emerald-505/20 to-emerald-950/25 border-emerald-500/20 text-emerald-350';
   391→    if (l.includes('biolog')) return 'from-amber-505/20 to-amber-950/25 border-amber-500/20 text-amber-350';
   392→    if (l.includes('math')) return 'from-pink-505/20 to-pink-950/25 border-pink-500/20 text-pink-350';
   393→    if (l.includes('ict')) return 'from-cyan-505/20 to-cyan-950/25 border-cyan-500/20 text-cyan-350';
   394→    return 'from-slate-800/10 to-slate-950/20 border-white/5 text-slate-350';
   395→  };
   396→
   397→  return (
   398→    <div
   399→      id="classroom-chat-root"
   400→      className="
   401→        w-full min-h-[600px] lg:h-[calc(100vh-180px)]
   402→        flex flex-col lg:flex-row
   403→        rounded-3xl border border-white/[0.06]
   404→        bg-slate-950/40 overflow-hidden
   405→        backdrop-blur-2xl relative shadow-2xl
   406→      "
   407→    >
   408→      {/* ───────────────────────────────────────────────────────────────────
   409→          CHANNELS SIDEBAR (left on desktop, full-screen on mobile when mobileView='channels')
   410→          ─────────────────────────────────────────────────────────────────── */}
   411→      <aside
   412→        className={`
   413→          w-full lg:w-80 xl:w-96 shrink-0
   414→          flex flex-col gap-3
   415→          bg-slate-950/70 border-r border-white/[0.04]
   416→          overflow-hidden
   417→          ${mobileView === 'channels' ? 'flex' : 'hidden lg:flex'}
   418→        `}
   419→      >
   420→        {/* Sidebar header */}
   421→        <div className="flex items-center justify-between px-4 py-3 border-b border-white/[0.04] shrink-0">
   422→          <div className="flex items-center gap-2 min-w-0">
   423→            <div className="p-1.5 rounded-lg bg-pink-500/10 border border-pink-500/20 shrink-0">
   424→              <Users className="w-3.5 h-3.5 text-pink-400" />
   425→            </div>
   426→            <span className="font-mono text-[10px] uppercase font-black tracking-widest text-slate-300 truncate">
   427→              {t('lobby')}
   428→            </span>
   429→          </div>
   430→          <span className="flex items-center gap-1 text-[9px] bg-emerald-500/10 text-emerald-400 border border-emerald-500/20 px-1.5 py-0.5 rounded font-mono font-bold">
   431→            <span className="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-pulse" />
   432→            {t('synced')}
   433→          </span>
   434→        </div>
   435→
   436→        {/* Active users strip */}
   437→        <div className="px-3 shrink-0">
   438→          <div className="bg-slate-900/40 border border-white/[0.03] p-2.5 rounded-2xl">
   439→            <div className="flex items-center justify-between mb-2">
   440→              <span className="text-[9px] font-bold text-slate-500 uppercase tracking-wider">{t('activeScholars')}</span>
   441→              <span className="text-[9px] font-mono text-cyan-400">{activeUsers.length} {t('onlineSuffix')}</span>
   442→            </div>
   443→            <div className="flex items-center gap-2.5 overflow-x-auto pb-1 scrollbar-none">
   444→              {activeUsers.length === 0 ? (
   445→                <span className="text-[10px] text-slate-500 italic px-1">Connecting…</span>
   446→              ) : (
   447→                activeUsers.slice(0, 8).map((item, idx) => (
   448→                  <div key={item.id || idx} className="relative group shrink-0 flex flex-col items-center gap-1" title={`${item.name} (${item.role})`}>
   449→                    <div className="relative">
   450→                      {item.profilePic ? (
   451→                        <img
   452→                          src={item.profilePic}
   453→                          alt={item.name}
   454→                          referrerPolicy="no-referrer"
   455→                          className="w-8 h-8 rounded-full object-cover border border-emerald-500/60 bg-slate-950"
   456→                        />
   457→                      ) : (
   458→                        <div className="w-8 h-8 rounded-full border border-emerald-500/60 bg-slate-900 flex items-center justify-center text-slate-200 text-[10px] font-bold">
   459→                          {item.name ? item.name.charAt(0).toUpperCase() : '?'}
   460→                        </div>
   461→                      )}
   462→                      <span className="absolute -bottom-0.5 -right-0.5 w-2.5 h-2.5 rounded-full bg-emerald-500 border-2 border-slate-950" />
   463→                    </div>
   464→                    <span className="text-[8px] text-slate-400 font-bold max-w-[44px] truncate leading-none text-center">
   465→                      {item.name ? item.name.split(' ')[0] : 'Scholar'}
   466→                    </span>
   467→                  </div>
   468→                ))
   469→              )}
   470→            </div>
   471→          </div>
   472→        </div>
   473→
   474→        {/* Channel list */}
   475→        <div className="flex-1 overflow-y-auto px-3 pb-3 space-y-1.5 scrollbar-thin">
   476→          <span className="block text-[9px] font-bold text-slate-500 uppercase tracking-widest px-2 pb-1 font-mono">
   477→            {t('chatRooms')}
   478→          </span>
   479→
   480→          {/* General channel */}
   481→          <button
   482→            onClick={() => {
   483→              setActiveChannelId('general');
   484→              setMobileView('chat');
   485→            }}
   486→            className={`w-full flex items-center gap-3 px-3 py-2.5 min-h-[48px] rounded-2xl text-xs font-semibold cursor-pointer transition-all border text-left ${
   487→              activeChannelId === 'general'
   488→                ? 'bg-gradient-to-r from-cyan-500/15 to-indigo-500/10 border-cyan-500/30 text-cyan-200'
   489→                : 'text-slate-400 hover:text-slate-200 hover:bg-white/[0.03] border-transparent'
   490→            }`}
   491→          >
   492→            <div className="p-2 rounded-xl bg-slate-800/60 border border-white/5 shrink-0">
   493→              <Hash className="w-3.5 h-3.5" />
   494→            </div>
   495→            <div className="min-w-0 flex-1">
   496→              <div className="font-bold truncate">General Lounge</div>
   497→              <div className="text-[9px] text-slate-500 truncate">Open discussion for everyone</div>
   498→            </div>
   499→          </button>
   500→
   501→          {/* Subject channels */}
   502→          {subjects.map((subj) => {
   503→            const isActive = activeChannelId === subj.id;
   504→            return (
   505→              <button
   506→                key={subj.id}
   507→                onClick={() => {
   508→                  setActiveChannelId(subj.id);
   509→                  setMobileView('chat');
   510→                }}
   511→                className={`w-full flex items-center gap-3 px-3 py-2.5 min-h-[48px] rounded-2xl text-xs font-semibold cursor-pointer transition-all border text-left ${
   512→                  isActive
   513→                    ? 'bg-gradient-to-r from-cyan-500/15 to-indigo-500/10 border-cyan-500/30 text-cyan-200'
   514→                    : 'text-slate-400 hover:text-slate-200 hover:bg-white/[0.03] border-transparent'
