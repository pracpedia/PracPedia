   509→  // Synthesize a beautiful, clean, luxury notification chime to alert students in a modern academy
   510→  const triggerHorrorScreamerSound = () => {
   511→    try {
   512→      const AudioContextClass = window.AudioContext || (window as any).webkitAudioContext;
   513→      if (!AudioContextClass) return;
   514→      const ctx = new AudioContextClass();
   515→
   516→      const playChime = (freq: number, delay: number, dur: number) => {
   517→        const osc = ctx.createOscillator();
   518→        const gain = ctx.createGain();
   519→        osc.type = "sine";
   520→        osc.frequency.setValueAtTime(freq, ctx.currentTime + delay);
   521→        gain.gain.setValueAtTime(0, ctx.currentTime + delay);
   522→        gain.gain.linearRampToValueAtTime(0.25, ctx.currentTime + delay + 0.04);
   523→        gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + delay + dur);
   524→        osc.connect(gain);
   525→        gain.connect(ctx.destination);
   526→        osc.start(ctx.currentTime + delay);
   527→        osc.stop(ctx.currentTime + delay + dur);
   528→      };
   529→
   530→      // Play a premium major-seventh ambient chord progression (C Major 7)
   531→      playChime(523.25, 0.0, 1.2); // C5
   532→      playChime(659.25, 0.08, 1.4); // E5
   533→      playChime(783.99, 0.16, 1.6); // G5
   534→      playChime(987.77, 0.24, 2.0); // B5
   535→    } catch (e) {
