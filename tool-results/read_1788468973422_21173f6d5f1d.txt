     1→'use client';
     2→
     3→import React, { useState, useEffect, useRef } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import { useAuth } from '@/contexts/AuthContext';
     6→import { safeLocalStorage } from '@/lib/storage';
     7→import { FormattedMarkdown } from './FormattedMarkdown';
     8→import {
     9→  ChevronLeft,
    10→  ChevronRight,
    11→  X,
    12→  ZoomIn,
    13→  ZoomOut,
    14→  Sparkles,
    15→  Cpu,
    16→  BookOpen,
    17→  MessageSquare,
    18→  Send,
    19→  HelpCircle,
    20→  FlaskConical,
    21→  RefreshCw,
    22→  CheckCircle2,
    23→  AlertTriangle,
    24→} from 'lucide-react';
    25→
    26→interface LightboxProps {
    27→  images: { url: string; title?: string }[];
    28→  initialIndex: number;
    29→  onClose: () => void;
    30→}
    31→
    32→export const Lightbox: React.FC<LightboxProps> = ({ images, initialIndex, onClose }) => {
    33→  const { apiFetch, user, setUser, geminiApiKey, setIsKeyModalOpen, language } = useAuth();
    34→
    35→  // Carousel states
    36→  const [index, setIndex] = useState(initialIndex);
    37→  const [zoom, setZoom] = useState(false);
    38→
    39→  // HSC Science AI Helper States
    40→  const [showAiAssistant, setShowAiAssistant] = useState(false);
    41→  const [isAnalyzing, setIsAnalyzing] = useState(false);
    42→  const [analysisResult, setAnalysisResult] = useState<string | null>(null);
    43→  const [errorMessage, setErrorMessage] = useState<string | null>(null);
    44→
    45→  // Custom Chat States
    46→  const [activeTab, setActiveTab] = useState<'guide' | 'chat'>('guide');
    47→  const [chatLogs, setChatLogs] = useState<Array<{ sender: 'student' | 'ai'; text: string }>>([]);
    48→  const [customQuestion, setCustomQuestion] = useState('');
    49→  const [isAnswering, setIsAnswering] = useState(false);
    50→  const [aiLanguage, setAiLanguage] = useState<'en' | 'bn_book'>(() => {
    51→    if (typeof window === 'undefined') return 'en';
    52→    return (safeLocalStorage.getItem('ai_language') as 'en' | 'bn_book') || 'en';
    53→  });
    54→
    55→  const handleLanguageChange = (lang: 'en' | 'bn_book') => {
    56→    setAiLanguage(lang);
    57→    safeLocalStorage.setItem('ai_language', lang);
    58→  };
    59→
    60→
    61→
    62→  // Automatic scroll container reference
    63→  const chatScrollRef = useRef<HTMLDivElement>(null);
    64→
    65→  useEffect(() => {
    66→    const container = chatScrollRef.current;
    67→    if (!container) return;
    68→
    69→    // Pin scroll position directly to bottom to prevent outer iframe scroll jitter
    70→    container.scrollTop = container.scrollHeight;
    71→
    72→    // Use ResizeObserver to auto-scroll when math blocks/LaTex images render asynchronously
    73→    const resizeObserver = new ResizeObserver(() => {
    74→      container.scrollTop = container.scrollHeight;
    75→    });
    76→
    77→    resizeObserver.observe(container);
    78→
    79→    // Fallback timer to double-check layout adjustments
    80→    const timer = setTimeout(() => {
    81→      container.scrollTop = container.scrollHeight;
    82→    }, 150);
    83→
    84→    return () => {
    85→      resizeObserver.disconnect();
    86→      clearTimeout(timer);
    87→    };
    88→  }, [chatLogs, isAnswering]);
    89→
    90→  // Reset state when swiping notebook sheets
    91→  const handleIndexChange = (newIdx: number) => {
    92→    setIndex(newIdx);
    93→    setZoom(false);
    94→    setAnalysisResult(null);
    95→    setChatLogs([]);
    96→    setErrorMessage(null);
    97→    setActiveTab('guide');
    98→  };
    99→
   100→  const handlePrev = (e?: React.MouseEvent) => {
   101→    e?.stopPropagation();
   102→    const newIdx = index > 0 ? index - 1 : images.length - 1;
   103→    handleIndexChange(newIdx);
   104→  };
   105→
   106→  const handleNext = (e?: React.MouseEvent) => {
   107→    e?.stopPropagation();
   108→    const newIdx = index < images.length - 1 ? index + 1 : 0;
   109→    handleIndexChange(newIdx);
   110→  };
   111→
   112→  const currentImage = images[index];
   113→
   114→  // Call backend parser
   115→  const handleAnalyzePage = async () => {
   116→    if (!currentImage?.url) return;
   117→
   118→    // ── BYOK gate — block AI features until the user connects a Gemini API key ──
   119→    if (!geminiApiKey || !geminiApiKey.trim()) {
   120→      setIsKeyModalOpen(true);
   121→      setErrorMessage(
   122→        language === 'bn'
   123→          ? 'এই ফিচারটি ব্যবহার করতে Gemini API কী সংযুক্ত করুন। আপনার নিজস্ব API কী দিন (BYOK) অথবা https://aistudio.google.com/app/apikey থেকে বিনামূল্যে একটি তৈরি করুন।'
   124→          : 'Connect your Gemini API key to use this feature. Bring Your Own Key (BYOK) — get a free one at https://aistudio.google.com/app/apikey'
   125→      );
   126→      return;
   127→    }
   128→
   129→    try {
   130→      setIsAnalyzing(true);
   131→      setErrorMessage(null);
   132→
   133→      const res = await apiFetch('/api/scan/analyze-page', {
   134→        method: 'POST',
   135→        headers: { 'Content-Type': 'application/json' },
   136→        body: JSON.stringify({ imageUrl: currentImage.url, language: aiLanguage })
   137→      });
   138→
   139→      if (!res.ok) {
   140→        let serverMsg = '';
   141→        try { const e = await res.json(); serverMsg = e.error || ''; } catch (_) {}
   142→        throw new Error(serverMsg || "Unable to read image scan. Check your internet connection.");
   143→      }
   144→
   145→      const data = await res.json();
   146→      setAnalysisResult(data.analysis);
   147→
   148→      setChatLogs([
   149→        {
   150→          sender: 'ai',
   151→          text: aiLanguage === 'bn_book'
   152→            ? `👋 আসসালামু আলাইকুম! আমি আপনার ব্যবহারিক নোটবুকের পৃষ্ঠাটি স্ক্যান করেছি। আমি এই অংশটির মূল তাত্ত্বিক ধারণা, সমীকরণ এবং রিডিং বুঝতে সাহায্য করতে প্রস্তুত। আপনি কি নিয়ে আলোচনা করতে চান?`
   153→            : `👋 Salam! I have scanned this practical notebook sheet. I am ready to explain the underlying theories, balance equations, or help you understand this table. What would you like to discuss?`
   154→        }
   155→      ]);
   156→    } catch (err: any) {
   157→      setErrorMessage(err.message || "Failed to establish connection with AI processing node.");
   158→    } finally {
   159→      setIsAnalyzing(false);
   160→    }
   161→  };
   162→
   163→  // Submit student chat questions
   164→  const handleAskQuestion = async (presetText?: string) => {
   165→    const textToSend = (presetText || customQuestion).trim();
   166→    if (!textToSend || !currentImage?.url) return;
   167→
   168→    // ── BYOK gate ──
   169→    if (!geminiApiKey || !geminiApiKey.trim()) {
   170→      setIsKeyModalOpen(true);
   171→      setErrorMessage(
   172→        language === 'bn'
   173→          ? 'এই ফিচারটি ব্যবহার করতে Gemini API কী সংযুক্ত করুন। আপনার নিজস্ব API কী দিন (BYOK) অথবা https://aistudio.google.com/app/apikey থেকে বিনামূল্যে একটি তৈরি করুন।'
   174→          : 'Connect your Gemini API key to use this feature. Bring Your Own Key (BYOK) — get a free one at https://aistudio.google.com/app/apikey'
   175→      );
   176→      return;
   177→    }
   178→
   179→    setChatLogs((prev) => [...prev, { sender: 'student', text: textToSend }]);
   180→    if (!presetText) setCustomQuestion('');
   181→
   182→    try {
   183→      setIsAnswering(true);
   184→      setErrorMessage(null);
   185→
   186→      const res = await apiFetch('/api/scan/analyze-page', {
   187→        method: 'POST',
   188→        headers: { 'Content-Type': 'application/json' },
   189→        body: JSON.stringify({
   190→          imageUrl: currentImage.url,
   191→          question: textToSend,
   192→          language: aiLanguage
   193→        })
   194→      });
   195→
   196→      if (!res.ok) {
   197→        throw new Error("Datalink failed to gather academic assessment.");
   198→      }
   199→
   200→      const data = await res.json();
   201→      setChatLogs((prev) => [...prev, { sender: 'ai', text: data.analysis }]);
   202→    } catch (err: any) {
   203→      setChatLogs((prev) => [...prev, { sender: 'ai', text: `⚠️ System Error: ${err.message || "Could not retrieve prompt reply."}` }]);
   204→    } finally {
   205→      setIsAnswering(false);
   206→    }
   207→  };
   208→
   209→  // Custom Markdown parser for rich formatted HSC typography
   210→  const renderHscMarkdownList = (text: string) => {
   211→    if (!text) return null;
   212→    const lines = text.split('\n');
   213→
   214→    return (
   215→      <div className="space-y-3.5 font-sans text-xs sm:text-[13px] leading-relaxed text-slate-300">
   216→        {lines.map((line, idx) => {
   217→          const trimmed = line.trim();
   218→          if (!trimmed) return <div key={idx} className="h-1" />;
   219→
   220→          // Headers formatting
   221→          if (trimmed.startsWith('# ')) {
   222→            return (
   223→              <h3 key={idx} className="text-sm sm:text-base font-black text-white mt-5 mb-2.5 pb-1 border-b border-white/5 flex items-center gap-2 break-words">
   224→                <span className="w-1 h-3.5 bg-gradient-to-b from-cyan-400 to-indigo-500 rounded-full shrink-0" />
   225→                {trimmed.substring(2)}
   226→              </h3>
   227→            );
   228→          }
   229→          if (trimmed.startsWith('## ') || trimmed.startsWith('### ')) {
   230→            const titleText = trimmed.startsWith('## ') ? trimmed.substring(3) : trimmed.substring(4);
   231→            return (
   232→              <h4 key={idx} className="text-xs sm:text-[12px] font-bold text-cyan-400 mt-4 mb-2 tracking-wide uppercase font-mono flex items-center gap-1.5 break-words">
   233→                <Sparkles className="w-3 h-3 text-cyan-400 animate-pulse shrink-0" />
   234→                {titleText}
   235→              </h4>
   236→            );
   237→          }
   238→
   239→          let content = trimmed;
   240→          const isBullet = trimmed.startsWith('- ') || trimmed.startsWith('* ');
   241→          if (isBullet) {
   242→            content = trimmed.substring(2);
   243→          }
   244→
   245→          // Simple code block display helper (formulas, fractions, etc.)
   246→          const isFormula = content.includes('$$') || content.includes('\\frac') || content.includes('\\pi');
   247→
   248→          // Bold parsing (**bold**)
   249→          const parts = content.split('**');
   250→          const renderedLine = parts.map((part, pIdx) => {
   251→            if (pIdx % 2 === 1) {
   252→              return <strong key={pIdx} className="text-amber-300 font-extrabold">{part}</strong>;
   253→            }
   254→            return <React.Fragment key={pIdx}>{part}</React.Fragment>;
   255→          });
   256→
   257→          if (isBullet) {
   258→            return (
   259→              <div key={idx} className="flex items-start gap-2 ml-1 mt-1 font-sans">
   260→                <span className="w-1.5 h-1.5 rounded-full bg-cyan-500 mt-2 shrink-0 animate-pulse" />
   261→                <div className="flex-1 text-slate-300 font-medium break-words">{renderedLine}</div>
   262→              </div>
   263→            );
   264→          }
   265→
   266→          if (isFormula) {
   267→            return (
   268→              <div key={idx} className="my-2.5 p-3 rounded-xl bg-slate-950/80 border border-cyan-900/30 text-center font-mono text-cyan-300 select-all overflow-x-auto text-[11px] leading-relaxed shadow-inner break-words">
   269→                {content.replace(/\$\$/g, '').trim()}
   270→              </div>
   271→            );
   272→          }
   273→
   274→          return <p key={idx} className="text-slate-350 font-normal leading-relaxed break-words">{renderedLine}</p>;
   275→        })}
   276→      </div>
   277→    );
   278→  };
   279→
   280→
   281→  return (
   282→    <AnimatePresence>
   283→      <motion.div
   284→        initial={{ opacity: 0 }}
   285→        animate={{ opacity: 1 }}
   286→        exit={{ opacity: 0 }}
   287→        onClick={(e) => {
   288→          if (e.target === e.currentTarget) onClose();
   289→        }}
   290→        className="fixed inset-0 z-50 flex flex-col lg:flex-row bg-[#030712]/98 backdrop-blur-xl select-none overflow-y-auto lg:overflow-hidden text-slate-100"
   291→      >
   292→
   293→        {/* ========================================================
   294→            LEFT PORTION: RESPONSIVE CAROUSEL & IMAGE VIEWER
   295→            ======================================================== */}
   296→        <div className="flex-1 flex flex-col justify-between p-3 sm:p-5 h-[48vh] sm:h-[50vh] lg:h-full relative overflow-hidden select-none border-b lg:border-b-0 lg:border-r border-white/5 bg-[#090d16]/30 min-h-0">
   297→
   298→          {/* Top Panel Header Actions */}
   299→          <div className="flex items-center justify-between w-full text-white/90 p-1.5 z-10 select-none shrink-0 gap-2">
   300→            <div className="flex flex-col text-left min-w-0">
   301→              <span className="text-[8px] font-mono tracking-widest text-cyan-400 font-black uppercase flex items-center gap-1">
   302→                <FlaskConical className="w-3 h-3 text-cyan-400 shrink-0" />
   303→                <span className="truncate">HSC NOTEBOOK SCANNER</span>
   304→              </span>
   305→              <span className="text-xs sm:text-sm font-black text-slate-150 truncate max-w-[140px] sm:max-w-md mt-0.5" title={currentImage?.title}>
   306→                {currentImage?.title || `Notebook Page ${index + 1}`}
   307→              </span>
   308→            </div>
   309→
   310→            <div className="flex items-center gap-1.5 sm:gap-2 shrink-0">
   311→              {/* HSC AI Toggle */}
   312→              <button
   313→                onClick={(e) => {
   314→                  e.stopPropagation();
   315→                  setShowAiAssistant(!showAiAssistant);
   316→                }}
   317→                className={`flex items-center gap-1.5 px-2 sm:px-3 py-2 min-h-[44px] rounded-xl text-[10px] font-black font-mono uppercase tracking-wide border-2 transition-all cursor-pointer ${
   318→                  showAiAssistant
   319→                    ? 'bg-cyan-500/10 border-cyan-400 text-cyan-300 animate-pulse shadow-[0_0_15px_rgba(34,211,238,0.15)]'
   320→                    : 'bg-white/5 border-white/10 hover:border-white/30 text-slate-300 hover:text-white'
   321→                }`}
   322→              >
   323→                <Cpu className="w-3.5 h-3.5 text-cyan-400 shrink-0" />
   324→                <span className="hidden sm:inline">AI Teacher</span>
   325→                <span className="sm:hidden">AI</span>
   326→              </button>
   327→
   328→              <span className="px-2.5 py-1.5 rounded-xl bg-white/5 border border-white/5 text-[9.5px] font-mono font-bold text-slate-400 shrink-0">
   329→                {index + 1} / {images.length}
   330→              </span>
   331→
   332→              <button
   333→                onClick={(e) => {
   334→                  e.stopPropagation();
   335→                  setZoom(!zoom);
   336→                }}
   337→                className="p-2 min-h-[44px] min-w-[44px] flex items-center justify-center rounded-xl bg-white/5 border border-white/10 text-slate-300 hover:text-white transition-colors cursor-pointer hidden md:flex"
   338→                title={zoom ? "Zoom Out" : "Zoom In"}
   339→              >
   340→                {zoom ? <ZoomOut className="w-4 h-4" /> : <ZoomIn className="w-4 h-4" />}
   341→              </button>
   342→
   343→              <button
   344→                onClick={onClose}
   345→                className="p-2 min-h-[44px] min-w-[44px] flex items-center justify-center rounded-xl bg-rose-500/10 border border-rose-500/20 hover:bg-rose-600 hover:border-transparent text-rose-300 hover:text-white transition-all cursor-pointer shadow-sm"
   346→                title="Exit viewer"
   347→              >
   348→                <X className="w-4 h-4" />
   349→              </button>
   350→            </div>
   351→          </div>
   352→
   353→          {/* Center Image viewport (extremely fit, non-distorting on any phone) */}
   354→          <div className="relative flex-1 flex items-center justify-center min-h-0 w-full select-none py-2 shrink-0">
   355→            {images.length > 1 && (
   356→              <button
   357→                onClick={handlePrev}
   358→                className="absolute left-1 sm:left-3 z-10 p-2 sm:p-3 min-h-[44px] min-w-[44px] flex items-center justify-center rounded-xl bg-slate-900/40 border border-white/5 hover:bg-white/10 text-white transition-all cursor-pointer active:scale-90"
   359→                aria-label="Previous image"
   360→              >
   361→                <ChevronLeft className="w-5 h-5" />
   362→              </button>
   363→            )}
   364→
   365→            <div
   366→              onClick={(e) => e.stopPropagation()}
   367→              className="w-full h-full flex items-center justify-center p-2 relative min-h-0"
   368→            >
   369→              <motion.img
   370→                key={index}
   371→                initial={{ scale: 0.97, opacity: 0 }}
   372→                animate={{
   373→                  scale: zoom ? 1.2 : 1,
   374→                  opacity: 1,
   375→                  cursor: zoom ? 'zoom-out' : 'zoom-in'
   376→                }}
   377→                exit={{ scale: 0.97, opacity: 0 }}
   378→                transition={{ type: "spring", damping: 30, stiffness: 220 }}
   379→                src={currentImage?.url}
   380→                alt={currentImage?.title}
   381→                onClick={() => setZoom(!zoom)}
   382→                className="max-w-[95vw] max-h-[34vh] sm:max-h-[38vh] lg:max-h-[72vh] object-contain rounded-2xl border border-white/10"
   383→              />
   384→            </div>
   385→
   386→            {images.length > 1 && (
   387→              <button
   388→                onClick={handleNext}
   389→                className="absolute right-1 sm:right-3 z-10 p-2 sm:p-3 min-h-[44px] min-w-[44px] flex items-center justify-center rounded-xl bg-slate-900/40 border border-white/5 hover:bg-white/10 text-white transition-all cursor-pointer active:scale-90"
   390→                aria-label="Next image"
   391→              >
   392→                <ChevronRight className="w-5 h-5" />
   393→              </button>
   394→            )}
   395→          </div>
   396→
   397→          {/* Bottom Thumbnails */}
   398→          <div className="w-full max-w-2xl mx-auto flex items-center justify-center gap-1.5 overflow-x-auto py-1 z-10 select-none shrink-0 pb-2">
   399→            {images.map((img, idx) => (
   400→              <button
   401→                key={idx}
   402→                onClick={(e) => {
   403→                  e.stopPropagation();
   404→                  handleIndexChange(idx);
   405→                }}
   406→                className={`relative h-8 w-11 rounded-lg overflow-hidden shrink-0 border-2 transition-all cursor-pointer ${
   407→                  idx === index ? 'border-cyan-400 scale-105 shadow-md shadow-cyan-500/20' : 'border-transparent opacity-40 hover:opacity-85'
   408→                }`}
   409→              >
   410→                <img src={img.url} className="w-full h-full object-cover" alt="" />
   411→              </button>
   412→            ))}
   413→          </div>
   414→        </div>
   415→
   416→
   417→        {/* ========================================================
   418→            RIGHT PORTION: MODERN, ROBUST HSC SCIENCE AI ASSISTANT
   419→            ======================================================== */}
   420→        <AnimatePresence>
   421→          {showAiAssistant ? (
   422→            <motion.div
   423→              initial={{ opacity: 0, x: typeof window !== 'undefined' && window.innerWidth < 1024 ? 0 : 360, y: typeof window !== 'undefined' && window.innerWidth < 1024 ? 120 : 0 }}
   424→              animate={{ opacity: 1, x: 0, y: 0 }}
   425→              exit={{ opacity: 0, x: typeof window !== 'undefined' && window.innerWidth < 1024 ? 0 : 360, y: typeof window !== 'undefined' && window.innerWidth < 1024 ? 120 : 0 }}
   426→              transition={{ type: "spring", damping: 28, stiffness: 200 }}
   427→              className="w-full lg:w-[480px] xl:w-[520px] bg-[#070b13] border-t lg:border-t-0 lg:border-l border-white/5 flex flex-col shrink-0 text-left h-[52vh] lg:h-full relative overflow-hidden z-20 select-text min-h-0"
   428→              onClick={(e) => e.stopPropagation()}
   429→            >
   430→
   431→              {/* Dynamic Header */}
   432→              <div className="p-3 sm:p-4 bg-[#0a101b] border-b border-white/5 flex items-center justify-between shrink-0 select-none gap-2">
   433→                <div className="flex items-center gap-2 sm:gap-3 min-w-0">
   434→                  <div className="relative shrink-0">
   435→                    <div className="absolute -inset-1 rounded-xl bg-cyan-500/30 blur animate-pulse" />
   436→                    <div className="relative p-2 rounded-xl bg-cyan-950/60 border border-cyan-500/35 text-cyan-300">
   437→                      <Sparkles className="w-4 h-4" />
   438→                    </div>
   439→                  </div>
   440→                  <div className="min-w-0">
   441→                    <h4 className="text-xs sm:text-sm font-black text-white tracking-wide truncate">Bangladesh HSC Practical Guide</h4>
   442→                    <span className="text-[9px] font-mono text-cyan-400 font-extrabold uppercase tracking-widest block mt-0.5 truncate">MULTIMODAL LESSON GENERATOR</span>
   443→                  </div>
   444→                </div>
   445→
   446→                <div className="flex items-center gap-1 bg-slate-950 border border-white/10 p-0.5 rounded-lg select-none shrink-0">
   447→                  <button
   448→                    onClick={() => handleLanguageChange('en')}
   449→                    className={`px-2 py-1 min-h-[28px] rounded text-[9px] font-black uppercase tracking-wider transition-all cursor-pointer ${aiLanguage === 'en' ? 'bg-gradient-to-r from-cyan-500 to-indigo-600 text-slate-950 font-black shadow-md' : 'text-slate-400 hover:text-slate-200'}`}
   450→                  >
   451→                    English
   452→                  </button>
   453→                  <button
