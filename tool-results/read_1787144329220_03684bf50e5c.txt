     2→

     4→import { useAuth } from '@/context/AuthContext';
     5→import { useLanguage } from '@/context/LanguageContext';
     6→import {
     7→  Send,

    31→
    32→interface SubjectType {
    33→  id: string;
    34→  title: string;
    35→  description: string;
    36→}
    37→
    38→interface MessageType {
    39→  id?: string;
    40→  _id?: string;
    41→  userId: string;
    42→  userName: string;
    43→  userRole: 'user' | 'admin';
    44→  content: string;
    45→  subjectId: string;
    46→  createdAt: string;
    47→  userProfilePic?: string;
    48→}
    49→
    50→interface ClassroomDiscussionProps {
    51→  subjects: SubjectType[];
    52→}
    53→
    54→export const ClassroomDiscussion: React.FC<ClassroomDiscussionProps> = ({ subjects }) => {
    55→  const { user, apiFetch } = useAuth();
    56→  const { language, t } = useLanguage();
    57→  const [messages, setMessages] = useState<MessageType[]>([]);
    58→  const [activeChannelId, setActiveChannelId] = useState<string>('general');
    59→  const [typedMessage, setTypedMessage] = useState<string>('');
    60→  const [loading, setLoading] = useState<boolean>(false);
    61→  const [submitting, setSubmitting] = useState<boolean>(false);
    62→  const [errorMessage, setErrorMessage] = useState<string | null>(null);
    63→
    64→  // Custom mobile screen flow: 'channels' list vs the 'chat' window itself
    65→  const [mobileView, setMobileView] = useState<'channels' | 'chat'>('chat');
    66→
    67→  const scrollContainerRef = useRef<HTMLDivElement>(null);
    68→
    69→  // Real-time active users (SSR-safe — window may be undefined during prerender)
    70→  const [activeUsers, setActiveUsers] = useState<any[]>(() =>
    71→    typeof window !== 'undefined' ? (window as any).__activeUsers || [] : [],
    72→  );
    73→
    74→  // Quick Action triggers for easy classroom conversation starter
    75→  const quickChips = [
    76→    t('quickChip1'),
    77→    t('quickChip2'),
    78→    t('quickChip3'),
    79→    t('quickChip4'),
    80→  ];
    81→
    82→  // Fetch messages of specified subject channel
    83→  const fetchChannelMessages = async (subjId: string, silent: boolean = false) => {
    84→    if (!silent) {
    85→      setLoading(true);
    86→      setErrorMessage(null);
    87→    }
    88→    try {
    89→      // Next.js backend exposes /api/chat (general) and /api/chat/[subjectId] (subject-specific)
    90→      const endpoint = subjId === 'general' ? '/api/chat' : `/api/chat/${subjId}`;
    91→      const res = await apiFetch(endpoint);
    92→      if (res.ok) {
    93→        const raw = await res.json();
    94→        // Normalize API fields (text -> content, userAvatar -> userProfilePic) to match local MessageType shape
    95→        const data: MessageType[] = (Array.isArray(raw) ? raw : []).map((m: any) => ({
    96→          id: m.id,
    97→          _id: m._id,
    98→          userId: m.userId,
    99→          userName: m.userName,
   100→          userRole: m.userRole === 'admin' ? 'admin' : 'user',
   101→          content: m.text ?? m.content ?? '',
   102→          subjectId: m.subjectId ?? subjId,
   103→          createdAt: m.createdAt,
   104→          userProfilePic: m.userProfilePic ?? m.userAvatar,
   105→        }));
   106→        setMessages((prev) => {
   107→          // Compare previous message IDs to prevent redundant state re-renders which mess up the layout
   108→          const prevIds = prev.map(m => m.id || m._id).join(',');
   109→          const newIds = data.map((m) => m.id || m._id).join(',');
   110→          if (prevIds !== newIds || prev.length !== data.length) {
   111→            return data;
   112→          }
   113→          return prev;
   114→        });
   115→      } else if (!silent) {
   116→        const errData = await res.json();
   117→        setErrorMessage(errData.error || t('networkLoss'));
   118→      }
   119→    } catch (e: any) {
   120→      if (!silent) {
   121→        setErrorMessage(t('networkLoss'));
   122→      }
   123→    } finally {
   124→      if (!silent) {
   125→        setLoading(false);
   126→      }
   127→    }
   128→  };
   129→
   130→  // Perform post of typed message content
   131→  const handleSendMessage = async (incomingContent?: string) => {
   132→    const targetContent = incomingContent || typedMessage;
   133→    if (!targetContent.trim() || submitting) return;
   134→
   135→    setSubmitting(true);
   136→    setErrorMessage(null);
   137→
   138→    const messageContent = targetContent;
   139→    if (!incomingContent) {
   140→      setTypedMessage('');
   141→    }
   142→
   143→    try {
   144→      // Next.js /api/chat expects { text, subjectId } — pass null for general lounge
   145→      const subjectPayload = activeChannelId === 'general' ? null : activeChannelId;
   146→      const res = await apiFetch('/api/chat', {
   147→        method: 'POST',
   148→        headers: {
   149→          'Content-Type': 'application/json',
   150→        },
   151→        body: JSON.stringify({
   152→          text: messageContent,
   153→          subjectId: subjectPayload
   154→        })
   155→      });
   156→
   157→      if (res.ok) {
   158→        const newMsg = await res.json();
   159→        // Normalize the new message to match local MessageType shape
   160→        const normalized: MessageType = {
   161→          id: newMsg.id,
   162→          _id: newMsg._id,
   163→          userId: newMsg.userId,
   164→          userName: newMsg.userName,
   165→          userRole: newMsg.userRole === 'admin' ? 'admin' : 'user',
   166→          content: newMsg.text ?? newMsg.content ?? messageContent,
   167→          subjectId: newMsg.subjectId ?? activeChannelId,
   168→          createdAt: newMsg.createdAt,
   169→          userProfilePic: newMsg.userProfilePic ?? newMsg.userAvatar,
   170→        };
   171→        setMessages((prev) => [...prev, normalized]);
   172→        // Force scroll-to-bottom on newly typed posts from our side
   173→        setTimeout(() => {
   174→          if (scrollContainerRef.current) {
   175→            scrollContainerRef.current.scrollTop = scrollContainerRef.current.scrollHeight;
   176→          }
   177→        }, 80);
   178→      } else {
   179→        const errData = await res.json();
   180→        setErrorMessage(errData.error || "Failed to broadcast message.");
   181→        if (!incomingContent) setTypedMessage(messageContent); // Restore text on failure
   182→      }
   183→    } catch (err) {
   184→      setErrorMessage("Broadcast failed due to structural network discrepancy.");
   185→      if (!incomingContent) setTypedMessage(messageContent); // Restore text on failure
   186→    } finally {
   187→      setSubmitting(false);
   188→    }
   189→  };
   190→
   191→  const handleFormSubmit = async (e: React.FormEvent) => {
   192→    e.preventDefault();
   193→    await handleSendMessage();
   194→  };
   195→
   196→  // Perform message eradication (admin or owner only)
   197→  const handleDeleteMessage = async (msgId: string) => {
   198→    if (!msgId) return;
   199→
   200→    try {
   201→      // Next.js /api/chat delete via ?id= query param
   202→      const res = await apiFetch(`/api/chat?id=${msgId}`, {
   203→        method: 'DELETE',
   204→      });
   205→
   206→      if (res.ok) {
   207→        // Remove locally from UI state
   208→        setMessages((prev) => prev.filter(m => (m.id || m._id) !== msgId));
   209→      } else {
   210→        const errData = await res.json();
   211→        alert(errData.error || "De-registration of message parameters refused.");
   212→      }
   213→    } catch (err) {
   214→      alert("Error deleting message. Check student permission rules.");
   215→    }
   216→  };
   217→
   218→  // Fetch initial posts and configure automatic polling every 1 second to make it act like a live Messenger
   219→  useEffect(() => {
   220→    fetchChannelMessages(activeChannelId, false);
   221→
   222→    const intervalId = setInterval(() => {
   223→      fetchChannelMessages(activeChannelId, true);
   224→    }, 1000);
   225→
   226→    return () => clearInterval(intervalId);
   227→  }, [activeChannelId]);
   228→
   229→  // Listen to WebSocket-triggered immediate reload event for instantaneous responsiveness
   230→  useEffect(() => {
   231→    const handleRemoteChatUpdate = (e: Event) => {
   232→      const customEvent = e as CustomEvent;
   233→      if (customEvent.detail && customEvent.detail.subjectId === activeChannelId) {
   234→        fetchChannelMessages(activeChannelId, true);
   235→      }
   236→    };
   237→
   238→    window.addEventListener('chat-updated', handleRemoteChatUpdate);
   239→    return () => {
   240→      window.removeEventListener('chat-updated', handleRemoteChatUpdate);
   241→    };
   242→  }, [activeChannelId]);
   243→
   244→  // Sync initial components presence data and update when event triggers
   245→  useEffect(() => {
   246→    if ((window as any).__activeUsers) {
   247→      setActiveUsers((window as any).__activeUsers);
   248→    }
   249→    const handlePresenceUpdated = (e: Event) => {
   250→      const customEvent = e as CustomEvent;
   251→      if (customEvent.detail && customEvent.detail.activeUsers) {
   252→        setActiveUsers(customEvent.detail.activeUsers);
   253→      }
   254→    };
   255→    window.addEventListener('presence-updated', handlePresenceUpdated);
   256→    return () => {
   257→      window.removeEventListener('presence-updated', handlePresenceUpdated);
   258→    };
   259→  }, []);
   260→
   261→  // Handle user-friendly quiet auto-scrolling on load or when sending message
   262→  const lastMessageCountRef = useRef<number>(0);
   263→  useEffect(() => {
   264→    const scrollContainer = scrollContainerRef.current;
   265→    if (!scrollContainer) return;
   266→
   267→    if (messages.length > lastMessageCountRef.current) {
   268→      const lastMsg = messages[messages.length - 1];
   269→      const isMyMessage = lastMsg && lastMsg.userId === user?.id;
   270→      const isInitialScroll = !loading && lastMessageCountRef.current === 0;
   271→
   272→      // Scroll to bottom immediately if it's the initial load or if the user sent a new message
   273→      if (isInitialScroll || isMyMessage) {
   274→        scrollContainer.scrollTop = scrollContainer.scrollHeight;
   275→      } else {
   276→        // If someone else posted, scroll down only if the user is already near the bottom
   277→        // (to prevent interrupting their reading of history)
   278→        const threshold = 180; // pixels from bottom threshold zone
   279→        const isNearBottom = scrollContainer.scrollHeight - scrollContainer.scrollTop - scrollContainer.clientHeight < threshold;
   280→        if (isNearBottom) {
   281→          scrollContainer.scrollTop = scrollContainer.scrollHeight;
   282→        }
   283→      }
   284→    }
   285→    lastMessageCountRef.current = messages.length;
   286→  }, [messages, loading, user?.id]);
   287→
   288→  const activeChannelObj = subjects.find(s => s.id === activeChannelId);
   289→  const activeChannelTitle = activeChannelId === 'general' ? 'General Academic Lounge' : `${activeChannelObj?.title} Support Channel`;
   290→  const activeChannelDesc = activeChannelId === 'general'
   291→    ? 'Universal classroom bulletin board and loose peer conversation space.'
   292→    : activeChannelObj?.description;
   293→
   294→  const formatDateLabel = (isoStr: string) => {
   295→    try {
   296→      const date = new Date(isoStr);
   297→      return date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
   298→    } catch (e) {
   299→      return '';
   300→    }
   301→  };
   302→
   303→  // Assign specific visual descriptors, emojis, and icons of subjects for professional feel
   304→  const getSubjectIcon = (title: string, sizeClass = "w-4 h-4") => {
   305→    const l = title.toLowerCase();
   306→    if (l.includes('physic')) return <Compass className={`${sizeClass} text-indigo-400`} />;
   307→    if (l.includes('chemist')) return <Beaker className={`${sizeClass} text-emerald-400`} />;
   308→    if (l.includes('biolog')) return <Sparkle className={`${sizeClass} text-amber-400`} />;
   309→    if (l.includes('math')) return <GraduationCap className={`${sizeClass} text-pink-400`} />;
   310→    if (l.includes('ict')) return <Code className={`${sizeClass} text-cyan-400`} />;
   311→    return <Hash className={`${sizeClass} text-slate-450`} />;
   312→  };
   313→
   314→  const getSubjectGradient = (title: string) => {
   315→    const l = title.toLowerCase();
   316→    if (l.includes('physic')) return 'from-indigo-505/20 to-indigo-950/25 border-indigo-500/20 text-indigo-350';
   317→    if (l.includes('chemist')) return 'from-emerald-505/20 to-emerald-950/25 border-emerald-500/20 text-emerald-350';
   318→    if (l.includes('biolog')) return 'from-amber-505/20 to-amber-950/25 border-amber-500/20 text-amber-350';
   319→    if (l.includes('math')) return 'from-pink-505/20 to-pink-950/25 border-pink-500/20 text-pink-350';
   320→    if (l.includes('ict')) return 'from-cyan-505/20 to-cyan-950/25 border-cyan-500/20 text-cyan-350';
   321→    return 'from-slate-800/10 to-slate-950/20 border-white/5 text-slate-350';
   322→  };
   323→
   324→  return (
   325→    <div id="classroom-chat-root" className="w-full h-[70vh] min-h-[520px] lg:h-[calc(100vh-175px)] flex flex-col lg:flex-row rounded-3xl border border-white/[0.05] bg-slate-950/40 overflow-hidden backdrop-blur-2xl relative shadow-2xl">
   326→
   327→      {/* Channels Sidebar Selection Block */}
   328→      {/* responsive class:hidden toggles visibility on mobile based on mobileView */}
   329→      <div className={`w-full lg:w-80 border-r border-white/[0.04] bg-slate-950/65 p-3 sm:p-4 shrink-0 flex flex-col gap-4 overflow-hidden h-full ${
   330→        mobileView === 'chat' ? 'hidden lg:flex' : 'flex'
   331→      }`}>
   332→        {/* Sidebar Title */}
   333→        <div className="flex items-center justify-between px-2 shrink-0 select-none">
   334→          <div className="flex items-center gap-2">
   335→            <Users className="w-4 h-4 text-pink-500" />
   336→            <span className="font-mono text-[10px] uppercase font-black tracking-widest text-slate-400">{t('lobby')}</span>
   337→          </div>
   338→          <span className="flex items-center gap-1.5 text-[9px] bg-emerald-555/10 text-emerald-450 border border-emerald-500/20 px-1.5 py-0.2 rounded font-mono font-black animate-pulse">
   339→            ● {t('synced')}
   340→          </span>
   341→        </div>        {/* MESSENGER ACTIVE PRESENCE ROW - HORIZONTAL AVATARS STRIP */}
   342→        <div className="bg-slate-900/15 border border-white/[0.03] p-3 rounded-2xl space-y-2 select-none shrink-0">
   343→          <div className="flex items-center gap-1.5 justify-between">
   344→            <span className="text-[9px] font-bold text-slate-500 uppercase tracking-wider font-sans">{t('activeScholars')}</span>
   345→            <span className="text-[8px] font-mono text-cyan-400">{activeUsers.length} {t('onlineSuffix')}</span>
   346→          </div>
   347→
   348→          <div className="flex items-center gap-3 overflow-x-auto pb-1 scrollbar-none">
   349→            {activeUsers.length === 0 ? (
   350→              <span className="text-[10px] text-slate-500 italic mt-1 px-1">Connecting...</span>
   351→            ) : (
   352→              activeUsers.map((item, idx) => (
   353→                <div key={item.id || idx} className="relative group shrink-0 flex flex-col items-center gap-1.5" title={`${item.name} (${item.role})`}>
   354→                  <div className="relative">
   355→                    {item.profilePic ? (
   356→                      <img
   357→                        src={item.profilePic}
   358→                        alt={item.name}
   359→                        referrerPolicy="no-referrer"
   360→                        className="w-9 h-9 rounded-full object-cover p-0.5 border border-emerald-500/80 shadow-[0_0_10px_rgba(16,185,129,0.2)] bg-slate-950"
   361→                      />
   362→                    ) : (
   363→                      <div className="w-9 h-9 rounded-full border border-emerald-500/80 bg-slate-900 flex items-center justify-center text-slate-200 text-xs font-black shadow-[0_0_10px_rgba(16,185,129,0.2)]">
   364→                        {item.name ? item.name.charAt(0).toUpperCase() : '?'}
   365→                      </div>
   366→                    )}
   367→                    <span className="absolute bottom-0 right-0 flex h-2.5 w-2.5">
   368→                      <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-emerald-400 opacity-75" />
   369→                      <span className="relative inline-flex rounded-full h-2.5 w-2.5 bg-emerald-500 border border-slate-950" />
   370→                    </span>
   371→                  </div>
   372→                  <span className="text-[8px] text-slate-400 font-mono tracking-tight font-extrabold max-w-[50px] truncate leading-none text-center">
   373→                    {item.name ? item.name.split(' ')[0] : 'Scholar'}
   374→                  </span>
   375→
   376→                  <div className="absolute top-11 left-1/2 -translate-x-1/2 bg-slate-950/95 border border-white/10 px-2.5 py-1.5 rounded-lg text-[10px] text-slate-200 opacity-0 group-hover:opacity-100 transition-opacity duration-200 pointer-events-none whitespace-nowrap z-50 shadow-xl">
   377→                    <span className="font-extrabold block text-slate-100">{item.name}</span>
   378→                    <span className="text-[8px] text-emerald-405 font-bold block uppercase mt-0.5 tracking-wider">● {item.role === 'admin' ? t('teacher') : t('student')}</span>
   379→                  </div>
   380→                </div>
   381→              ))
   382→            )}
   383→          </div>
   384→        </div>
   385→
   386→        {/* Chats Channels List Container */}
   387→        <div className="flex-1 overflow-y-auto space-y-1.5 pr-1 select-none scrollbar-thin">
   388→          <span className="block text-[9px] font-bold text-slate-500 uppercase tracking-widest px-2 pb-1 font-mono">{t('chatRooms')}</span>
   389→
   390→          {/* General Lounge Channel button */}
   391→          <button
   392→            onClick={() => {
   393→              setActiveChannelId('general');
   394→              setMobileView('chat');
   395→            }}
   396→            className={`w-full flex items-center gap-3 px-3 py-3 min-h-[44px] rounded-2xl text-xs font-semibold cursor-pointer transition-all duration-300 border text-left ${
   397→              activeChannelId === 'general'
   398→                ? 'bg-gradient-to-r from-cyan-600/10 to-indigo-600/10 border-cyan-500/25 text-cyan-300 shadow-md shadow-cyan-950/20'
   399→                : 'text-slate-400 hover:text-slate-200 hover:bg-white/[0.02] border-transparent'
   400→            }`}
   401→          >
   402→            <div className={`p-2 rounded-xl shrink-0 ${
   403→              activeChannelId === 'general' ? 'bg-cyan-500/15' : 'bg-slate-900/80 border border-white/5'
   404→            }`}>
   405→              <MessageSquare className="w-4 h-4 text-cyan-400" />
   406→            </div>
   407→
   408→            <div className="min-w-0 flex-1 leading-snug">
   409→              <div className="flex items-center justify-between">
   410→                <span className="font-extrabold text-slate-200 truncate">{t('generalLounge')}</span>
   411→                <span className="text-[8px] font-mono text-slate-500">Global</span>
   412→              </div>
   413→              <p className="text-[9px] text-slate-510 truncate mt-0.5 font-medium">{t('generalLoungeDesc')}</p>
   414→            </div>
   415→          </button>
   416→
   417→          {/* Subjects channels mapped */}
   418→          {subjects.map((sub) => {
   419→            const isActive = activeChannelId === sub.id;
   420→            const gradClass = getSubjectGradient(sub.title);
   421→
   422→            return (
   423→              <button
   424→                key={sub.id}
   425→                onClick={() => {
   426→                  setActiveChannelId(sub.id);
   427→                  setMobileView('chat');
   428→                }}
   429→                className={`w-full flex items-center gap-3 px-3 py-3 min-h-[44px] rounded-2xl text-xs font-semibold cursor-pointer transition-all duration-300 border text-left ${
   430→                  isActive
   431→                    ? 'bg-gradient-to-r ' + gradClass + ' font-extrabold shadow-md'
   432→                    : 'text-slate-400 hover:text-slate-200 hover:bg-white/[0.02] border-transparent'
   433→                }`}
   434→              >
   435→                <div className={`p-2 rounded-xl shrink-0 ${
   436→                  isActive ? 'bg-white/10' : 'bg-slate-900/80 border border-white/5'
   437→                }`}>
   438→                  {getSubjectIcon(sub.title, "w-4 h-4")}
   439→                </div>
   440→
   441→                <div className="min-w-0 flex-1 leading-snug">
   442→                  <div className="flex items-center justify-between">
   443→                    <span className="font-extrabold text-slate-200 truncate">{sub.title} Channel</span>
   444→                    <span className="inline-block w-1.5 h-1.5 rounded-full bg-emerald-500" />
   445→                  </div>
   446→                  <p className="text-[9px] text-slate-510 truncate mt-0.5 font-medium">{sub.description}</p>
   447→                </div>
   448→              </button>
   449→            );
   450→          })}
   451→        </div>
   452→
   453→        {/* Bottom Sidebar Arabic calligraphy quote */}
   454→        <div className="hidden lg:flex flex-col items-center justify-center p-4 rounded-2xl border border-white/[0.03] bg-gradient-to-b from-slate-950 to-slate-950/60 shrink-0 text-center space-y-1 select-none">
   455→          <span className="text-sm font-bold bg-gradient-to-r from-amber-400 via-rose-300 to-indigo-400 bg-clip-text text-transparent font-arabic" style={{ direction: 'rtl' }}>
   456→            رَبِّ زِدْنِي عِلْمًا
   457→          </span>
   458→          <span className="text-[9px] font-mono font-bold text-slate-500 uppercase tracking-widest leading-none">
   459→            Rabbi Zidni Ilma
   460→          </span>
   461→        </div>
   462→      </div>
   463→
   464→      {/* Main Messaging Container Area */}
   465→      {/* responsive class:hidden toggles visibility on mobile based on mobileView */}
   466→      <div className={`flex-1 flex flex-col justify-between overflow-hidden relative h-full ${
   467→        mobileView === 'channels' ? 'hidden lg:flex' : 'flex'
   468→      }`}>
   469→
   470→        {/* Dynamic header of the active channel */}
   471→        <div className="bg-slate-950/75 border-b border-white/[0.04] px-3 sm:px-4 py-3 sm:py-3.5 flex items-center justify-between shrink-0 select-none">
   472→          <div className="flex items-center gap-2 sm:gap-3 min-w-0">
   473→            {/* Mobile Channel Back Navigation Arrow */}
   474→            <button
   475→              onClick={() => setMobileView('channels')}
   476→              className="lg:hidden p-2 min-h-[44px] min-w-[44px] flex items-center justify-center rounded-xl bg-slate-900 border border-white/5 hover:bg-slate-800 text-slate-300 cursor-pointer transition-colors shrink-0"
   477→              title="Show Lobby"
