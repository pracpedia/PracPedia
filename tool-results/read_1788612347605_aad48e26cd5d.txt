   117→  // Draggable corners calculations
   118→  const handleControlPointDrag = (corner: 'tl'|'tr'|'br'|'bl', clientX: number, clientY: number) => {
   119→    const container = containerRef.current;
   120→    if (!container) return;

   128→      ...prev,
   129→      [corner]: { x, y }

   139→      const clientY = 'touches' in moveEvent ? moveEvent.touches[0].clientY : moveEvent.clientY;
   140→      handleControlPointDrag(corner, clientX, clientY);

   142→
   143→    const stopHandler = () => {
   144→      document.removeEventListener('mousemove', moveHandler);
   145→      document.removeEventListener('mouseup', stopHandler);
   146→      document.removeEventListener('touchmove', moveHandler);
   147→      document.removeEventListener('touchend', stopHandler);
   148→    };
   149→
   150→    document.addEventListener('mousemove', moveHandler);
   151→    document.addEventListener('mouseup', stopHandler);
   152→    document.addEventListener('touchmove', moveHandler, { passive: false });
   153→    document.addEventListener('touchend', stopHandler);

   156→  // Gemini AI auto corner detection tool request call
   157→  const triggerAICornerDetection = async () => {
   158→    if (!rawImageBase64) return;

   160→    // ── BYOK gate ──
   161→    if (!geminiApiKey || !geminiApiKey.trim()) {
   162→      setIsKeyModalOpen(true);
   163→      setError(
   164→        language === 'bn'

   166→          : 'Connect your Gemini API key to use this feature. Bring Your Own Key (BYOK) — get a free one at https://aistudio.google.com/app/apikey'
   167→      );

   172→    setError(null);
   173→    try {
   174→      const res = await apiFetch('/api/scan/detect-corners', {
   175→        method: 'POST',
   176→        headers: { 'Content-Type': 'application/json' },
   177→        body: JSON.stringify({ imageBase64: rawImageBase64 })
   178→      });
   179→
   180→      const data = await res.json();
   181→      if (!res.ok) {
   182→        throw new Error(data.error || "Failed to trigger AI vision assistance.");
   183→      }
   184→
   185→      if (data.corners) {
   186→        setCorners({
   187→          tl: { x: data.corners.topLeft[0], y: data.corners.topLeft[1] },
   188→          tr: { x: data.corners.topRight[0], y: data.corners.topRight[1] },
   189→          br: { x: data.corners.bottomRight[0], y: data.corners.bottomRight[1] },
   190→          bl: { x: data.corners.bottomLeft[0], y: data.corners.bottomLeft[1] }
   191→        });
   192→        setSuccessMsg("AI scan boundaries locked successfully!");
   193→        setTimeout(() => setSuccessMsg(null), 1500);
   194→      }
   195→    } catch (err: any) {
   196→      console.warn("AI corner detection fallback triggered:", err.message);
   197→      setError("AI model busy, approx boundaries placed. Feel free to drag the glowing corners manually!");
   198→    } finally {
   199→      setIsProcessing(false);
   200→    }
   201→  };
   202→
   203→  const handleResetCorners = () => {
   204→    setCorners({
   205→      tl: { x: 0.10, y: 0.10 },
   206→      tr: { x: 0.90, y: 0.10 },
   207→      br: { x: 0.90, y: 0.90 },
   208→      bl: { x: 0.10, y: 0.90 }
   209→    });
   210→  };
   211→
   212→  // Core perspective transformation & image filter flattening engine
   213→  const executePerspectiveAlign = () => {
   214→    if (!hiddenImgRef.current || !rawImageBase64) return;
   215→    setIsProcessing(true);
   216→
   217→    setTimeout(() => {
   218→      try {
   219→        const img = hiddenImgRef.current!;
   220→        const srcW = img.naturalWidth || img.width;
   221→        const srcH = img.naturalHeight || img.height;
   222→
   223→        const srcCanvas = document.createElement('canvas');
   224→        srcCanvas.width = srcW;
   225→        srcCanvas.height = srcH;
   226→        const srcCtx = srcCanvas.getContext('2d');
   227→        if (!srcCtx) {
   228→          setError("Canvas processor failed to execute context.");
   229→          setIsProcessing(false);
   230→          return;
   231→        }
   232→
   233→        srcCtx.drawImage(img, 0, 0);
   234→        const srcData = srcCtx.getImageData(0, 0, srcW, srcH);
   235→
   236→        // Calculate original high-resolution corner coords in pixels
   237→        const x0 = corners.tl.x * srcW, y0 = corners.tl.y * srcH;
   238→        const x1 = corners.tr.x * srcW, y1 = corners.tr.y * srcH;
   239→        const x2 = corners.br.x * srcW, y2 = corners.br.y * srcH;
   240→        const x3 = corners.bl.x * srcW, y3 = corners.bl.y * srcH;
   241→
   242→        // Calculate width and height dynamically based on the actual physical quadrilaterals (no shrinking)
   243→        const topWidth = Math.hypot(x1 - x0, y1 - y0);
   244→        const bottomWidth = Math.hypot(x2 - x3, y2 - y3);
   245→        const leftHeight = Math.hypot(x3 - x0, y3 - y0);
   246→        const rightHeight = Math.hypot(x2 - x1, y2 - y1);
   247→
   248→        let dstW = Math.round(Math.max(topWidth, bottomWidth));
   249→        let dstH = Math.round(Math.max(leftHeight, rightHeight));
   250→
   251→        // Sanitize dimension calculation boundaries
   252→        if (dstW < 300) dstW = 800;
   253→        if (dstH < 300) dstH = 1100;
   254→
   255→        // Safe upper bound limit for mobile browsers processing memory
   256→        const maxLimit = 2500;
   257→        if (dstW > maxLimit || dstH > maxLimit) {
   258→          const ratio = dstW / dstH;
   259→          if (dstW > dstH) {
   260→            dstW = maxLimit;
   261→            dstH = Math.round(maxLimit / ratio);
   262→          } else {
   263→            dstH = maxLimit;
   264→            dstW = Math.round(maxLimit * ratio);
   265→          }
   266→        }
   267→
   268→        setScannedWidth(dstW);
   269→        setScannedHeight(dstH);
   270→
   271→        const dstCanvas = document.createElement('canvas');
   272→        dstCanvas.width = dstW;
   273→        dstCanvas.height = dstH;
   274→        const dstCtx = dstCanvas.getContext('2d');
   275→        if (!dstCtx) {
   276→          setError("Destination canvas allocation failed.");
   277→          setIsProcessing(false);
   278→          return;
   279→        }
   280→
   281→        const dstData = dstCtx.createImageData(dstW, dstH);
   282→        const srcPixels = srcData.data;
   283→        const dstPixels = dstData.data;
   284→
   285→        for (let yd = 0; yd < dstH; yd++) {
   286→          const v = yd / (dstH - 1);
   287→          for (let xd = 0; xd < dstW; xd++) {
   288→            const u = xd / (dstW - 1);
   289→
   290→            // Pure bilinear interpolation mapping
   291→            const xs = Math.round((1 - u) * (1 - v) * x0 + u * (1 - v) * x1 + u * v * x2 + (1 - u) * v * x3);
   292→            const ys = Math.round((1 - u) * (1 - v) * y0 + u * (1 - v) * y1 + u * v * y2 + (1 - u) * v * y3);
   293→
   294→            const clampedXs = Math.max(0, Math.min(srcW - 1, xs));
   295→            const clampedYs = Math.max(0, Math.min(srcH - 1, ys));
   296→
   297→            const srcIdx = (clampedYs * srcW + clampedXs) * 4;
   298→            const dstIdx = (yd * dstW + xd) * 4;
   299→
   300→            let r = srcPixels[srcIdx];
   301→            let g = srcPixels[srcIdx + 1];
   302→            let b = srcPixels[srcIdx + 2];
   303→            const a = srcPixels[srcIdx + 3];
   304→
   305→            if (activeFilter === 'bw') {
   306→              // Document scanning filter (convert grayscale + expand contrast stretch + adaptively wash shadows)
   307→              const grayscale = 0.299 * r + 0.587 * g + 0.114 * b;
   308→
   309→              let finalVal = grayscale;
   310→              if (grayscale > 115) {
   311→                // Wash out soft lighting reflections to clean white background
   312→                finalVal = Math.min(255, grayscale * 1.35);
   313→              } else {
   314→                // Deepen pen/pencil ink contrasts
   315→                finalVal = Math.max(0, grayscale * 0.75);
   316→              }
   317→
   318→              // Contrast boosting stretch formula
   319→              finalVal = (finalVal - 128) * 1.6 + 128;
   320→              finalVal = Math.max(0, Math.min(255, finalVal));
   321→
   322→              r = finalVal;
   323→              g = finalVal;
   324→              b = finalVal;
   325→            } else if (activeFilter === 'color') {
   326→              // Enhanced color scan to suppress soft background shadows
   327→              r = Math.min(255, r * 1.18);
   328→              g = Math.min(255, g * 1.18);
   329→              b = Math.min(255, b * 1.18);
   330→            }
   331→
   332→            dstPixels[dstIdx] = r;
   333→            dstPixels[dstIdx + 1] = g;
   334→            dstPixels[dstIdx + 2] = b;
   335→            dstPixels[dstIdx + 3] = a;
   336→          }
   337→        }
   338→
   339→        dstCtx.putImageData(dstData, 0, 0);
   340→        const dataUrl = dstCanvas.toDataURL('image/jpeg', 0.90);
   341→        setScannedResultBase64(dataUrl);
   342→        setIsProcessing(false);
   343→        setStep(3); // Progress to name save page
   344→      } catch (err: any) {
   345→        console.warn("Canvas workspace alignment error context fallback activated:", err);
   346→        // Fallback: Bypass warping/cropping filters and use original raw image
   347→        setScannedResultBase64(rawImageBase64);
   348→        setIsProcessing(false);
   349→        setStep(3);
   350→      }
   351→    }, 100);
   352→  };

   354→  // Convert Base64 scan output to actual JPEG file blob to upload securely
   355→  const submitAlignedPageUpload = async (e: React.FormEvent) => {
   356→    e.preventDefault();
   357→    if (!scannedResultBase64) return;
   358→    setError(null);
   359→    setSuccessMsg(null);
   360→    setIsProcessing(true);
   361→
   362→    const checkTitle = title.trim() || 'Scanned Notebook Page';
   363→
   364→    try {
   365→      // Decode Base64 string to Blob
   366→      const b64Data = scannedResultBase64.replace(/^data:image\/\w+;base64,/, "");
   367→      const byteCharacters = atob(b64Data);
   368→      const byteArrays: BlobPart[] = [];
   369→      for (let offset = 0; offset < byteCharacters.length; offset += 512) {
   370→        const slice = byteCharacters.slice(offset, offset + 512);
   371→        const byteNumbers = new Array(slice.length);
   372→        for (let i = 0; i < slice.length; i++) {
   373→          byteNumbers[i] = slice.charCodeAt(i);
   374→        }
   375→        const byteArray = new Uint8Array(byteNumbers);
   376→        byteArrays.push(byteArray);
   377→      }
   378→      const blob = new Blob(byteArrays, { type: 'image/jpeg' });
   379→
   380→      // Package into standard Multipart FormData
   381→      const formData = new FormData();
   382→      formData.append('image', blob, 'scanned_notebook_page.jpg');
   383→
   384→      // 1. Upload scan image to backend uploads folder with secure validation
   385→      const uploadRes = await apiFetch('/api/images/upload-file', {
   386→        method: 'POST',
   387→        body: formData,
   388→      });
   389→
   390→      const uploadResult = await uploadRes.json();
   391→      if (!uploadRes.ok) {
   392→        throw new Error(uploadResult.error || "Fail to upload parsed scan image.");
   393→      }
   394→
   395→      // 2. Map scanned image URL coordinates to Course Practical Folder
   396→      await onUploadSuccess(uploadResult.url, checkTitle);
   397→
   398→      setSuccessMsg("Notebook scan file attached successfully!");
   399→      setTimeout(() => {
   400→        onClose();
   401→        resetModal();
   402→      }, 1200);
   403→    } catch (err: any) {
   404→      setError(err.message || "Workbook scan upload failed.");
   405→    } finally {
   406→      setIsProcessing(false);
   407→    }
   408→  };
   409→

   413→    if (!title.trim()) {
   414→      setError("Please input a valid page description label.");
   415→      return;
   416→    }
   417→    setError(null);
   418→    setIsProcessing(true);
   419→
   420→    try {
   421→      const checkTitle = title.trim();
   422→      await onUploadSuccess(rawImageBase64?.trim() || "", checkTitle);
   423→      setSuccessMsg("External web notebook link attached!");
   424→      setTimeout(() => {
   425→        onClose();
   426→        resetModal();
   427→      }, 1200);
   428→    } catch (err: any) {
   429→      setError(err.message || "Failed to link external path.");
   430→    } finally {
   431→      setIsProcessing(false);
   432→    }
   433→  };
   434→
   435→  const resetModal = () => {
   436→    setStep(1);
   437→    setRawImageBase64(null);
   438→    setScannedResultBase64(null);
   439→    setScannedWidth(0);
   440→    setScannedHeight(0);
   441→    setImageLoaded(false);
   442→    setTitle('');
   443→    setError(null);
   444→    setSuccessMsg(null);
   445→    setCorners({
   446→      tl: { x: 0.15, y: 0.15 },
   447→      tr: { x: 0.85, y: 0.15 },
   448→      br: { x: 0.85, y: 0.85 },
   449→      bl: { x: 0.15, y: 0.85 }
   450→    });
   451→  };
   452→
   453→  return (
   454→    <AnimatePresence>
   455→      <div className="fixed inset-0 z-50 flex items-center justify-center p-3 sm:p-4 overflow-y-auto">
   456→        {/* Backdrop glass blur overlay */}
   457→        <motion.div
   458→          initial={{ opacity: 0 }}
   459→          animate={{ opacity: 1 }}
   460→          exit={{ opacity: 0 }}
   461→          onClick={() => { onClose(); resetModal(); }}
   462→          className="absolute inset-0 bg-black/75 backdrop-blur-[6px]"
   463→          id="scanner-modal-backdrop"
   464→        />
   465→
   466→        {/* Floating Scanner Box */}
   467→        <motion.div
   468→          initial={{ scale: 0.95, y: 25, opacity: 0 }}
   469→          animate={{ scale: 1, y: 0, opacity: 1 }}
   470→          exit={{ scale: 0.95, y: 25, opacity: 0 }}
   471→          className="relative w-full max-w-2xl max-w-[92vw] bg-zinc-950 border border-zinc-800 rounded-3xl overflow-hidden shadow-2xl flex flex-col max-h-[92vh] text-slate-100 my-4"
   472→          id="scanner-container-card"
   473→        >
   474→          {/* Header segment banner */}
   475→          <div className="p-4 sm:p-5 border-b border-zinc-850 bg-zinc-90 w-full flex items-center justify-between gap-3">
   476→            <div className="flex items-center gap-3 min-w-0">
   477→              <div className="p-2.5 rounded-2xl bg-cyan-500/10 text-cyan-400 border border-cyan-500/20 shadow-md shrink-0">
   478→                <FileImage className="w-5 h-5" />
   479→              </div>
   480→              <div className="min-w-0">
   481→                <h3 className="text-sm font-black text-white flex items-center gap-2 tracking-tight flex-wrap">
   482→                  <span className="truncate">Advanced Notebook Scanner</span>
   483→                  <span className="text-[9px] bg-cyan-400/10 text-cyan-400 px-2 py-0.5 rounded-full font-mono border border-cyan-400/20 uppercase tracking-widest font-extrabold animate-pulse shrink-0">Auto Align</span>
   484→                </h3>
   485→                <p className="text-[10px] text-zinc-500 font-mono tracking-wide mt-0.5 uppercase truncate max-w-[200px] sm:max-w-[320px]">
   486→                  Subject Folder: {folderTitle}
   487→                </p>
   488→              </div>
   489→            </div>
   490→
   491→            <button
   492→              onClick={() => { onClose(); resetModal(); }}
   493→              className="p-1.5 min-h-[36px] min-w-[36px] flex items-center justify-center rounded-lg text-zinc-400 hover:text-white hover:bg-zinc-800/50 transition-all cursor-pointer border border-transparent hover:border-zinc-800 shrink-0"
   494→              aria-label="Close"
   495→            >
   496→              <X className="w-4 h-4" />
   497→            </button>
   498→          </div>
   499→
   500→          {/* Core Content Body wizard switcher */}
   501→          <div className="flex-1 overflow-y-auto p-4 sm:p-5 space-y-4 min-h-0">
   502→
   503→            {/* Notifications / Errors banner */}
   504→            {error && (
   505→              <div className="p-3 bg-rose-500/10 border border-rose-500/20 text-rose-400 text-xs font-semibold rounded-xl flex items-center gap-2.5 break-words">
   506→                <AlertTriangle className="w-4 h-4 shrink-0" />
   507→                <span className="flex-1 text-[11px] leading-relaxed break-words">{error}</span>
   508→              </div>
   509→            )}
   510→
   511→            {successMsg && (
   512→              <div className="p-3 bg-emerald-500/10 border border-emerald-500/20 text-emerald-400 text-xs font-semibold rounded-xl flex items-center gap-2.5 break-words">
   513→                <CheckCircle className="w-4 h-4 shrink-0 animate-bounce" />
   514→                <span className="text-[11px] font-bold break-words">{successMsg}</span>
   515→              </div>
   516→            )}
   517→
   518→            {/* STEP 1: CHOOSE AND INGEST PHOTO SOURCE */}
   519→            {step === 1 && (
   520→              <div className="space-y-4">
