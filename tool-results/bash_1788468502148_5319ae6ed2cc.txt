   508→  // Synthesize a beautiful, clean, luxury notification chime to alert students in a modern academy
   509→  const triggerHorrorScreamerSound = () => {
   510→    try {
   511→      const AudioContextClass = window.AudioContext || (window as any).webkitAudioContext;
   512→      if (!AudioContextClass) return;
   513→      const ctx = new AudioContextClass();
   514→
   515→      const playChime = (freq: number, delay: number, dur: number) => {
   516→        const osc = ctx.createOscillator();
   517→        const gain = ctx.createGain();
   518→        osc.type = "sine";
   519→        osc.frequency.setValueAtTime(freq, ctx.currentTime + delay);
   520→        gain.gain.setValueAtTime(0, ctx.currentTime + delay);
   521→        gain.gain.linearRampToValueAtTime(0.25, ctx.currentTime + delay + 0.04);
   522→        gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + delay + dur);
   523→        osc.connect(gain);
   524→        gain.connect(ctx.destination);
   525→        osc.start(ctx.currentTime + delay);
   526→        osc.stop(ctx.currentTime + delay + dur);
   527→      };
   528→
   529→      // Play a premium major-seventh ambient chord progression (C Major 7)
   530→      playChime(523.25, 0.0, 1.2); // C5
   531→      playChime(659.25, 0.08, 1.4); // E5
   532→      playChime(783.99, 0.16, 1.6); // G5
   533→      playChime(987.77, 0.24, 2.0); // B5
   534→    } catch (e) {
   535→      // Silent — browsers block AudioContext until user gesture (autoplay policy).
   536→      // This is expected behavior, not a bug, so we don't pollute the console.
   537→      void e;
   538→    }
