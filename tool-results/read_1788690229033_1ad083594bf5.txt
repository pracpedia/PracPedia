    34→
    35→interface SubjectType {
    36→  id: string;
    37→  title: string;
    38→  description: string;
    39→}
    40→
    41→interface MessageType {
    42→  id?: string;
    43→  _id?: string;
    44→  userId: string;
    45→  userName: string;
    46→  userRole: 'user' | 'admin';
    47→  content: string;
    48→  subjectId: string;
    49→  createdAt: string;
    50→  userProfilePic?: string;
    51→  imageUrl?: string;
    52→}
    53→
    54→interface ClassroomDiscussionProps {
    55→  subjects: SubjectType[];
    56→}
    57→
    58→export const ClassroomDiscussion: React.FC<ClassroomDiscussionProps> = ({ subjects }) => {
    59→  const { user, apiFetch } = useAuth();
    60→  const { language, t } = useLanguage();
    61→  const [messages, setMessages] = useState<MessageType[]>([]);
    62→  const messagesRef = useRef<MessageType[]>([]);
    63→  const [activeChannelId, setActiveChannelId] = useState<string>('general');
    64→  const [typedMessage, setTypedMessage] = useState<string>('');
    65→  const [pendingImage, setPendingImage] = useState<string | null>(null);
    66→  const [zoomedImage, setZoomedImage] = useState<string | null>(null);
    67→  const [revealedTimeId, setRevealedTimeId] = useState<string | null>(null);
    68→  const imageInputRef = useRef<HTMLInputElement>(null);
    69→  const [loading, setLoading] = useState<boolean>(false);
    70→  const [submitting, setSubmitting] = useState<boolean>(false);
    71→  const [errorMessage, setErrorMessage] = useState<string | null>(null);
    72→
    73→  // Close zoom popup on Escape
    74→  useEffect(() => {
    75→    if (!zoomedImage) return;
    76→    const handler = (e: KeyboardEvent) => {
    77→      if (e.key === 'Escape') setZoomedImage(null);
    78→    };
    79→    window.addEventListener('keydown', handler);
    80→    return () => window.removeEventListener('keydown', handler);
    81→  }, [zoomedImage]);
    82→
    83→  // Custom mobile screen flow: 'channels' list vs the 'chat' window itself
    84→  const [mobileView, setMobileView] = useState<'channels' | 'chat'>('chat');
    85→
    86→  const scrollContainerRef = useRef<HTMLDivElement>(null);
    87→
    88→  // Keep messagesRef in sync with messages state so the long-poll loop can
    89→  // read the latest message timestamp without being a dependency of the effect.
    90→  useEffect(() => {
    91→    messagesRef.current = messages;
    92→  }, [messages]);
    93→
    94→  // Real-time active users — guard against SSR (window is undefined on server)
    95→  const [activeUsers, setActiveUsers] = useState<any[]>(() => {
    96→    if (typeof window === 'undefined') return [];
    97→    return (window as any).__activeUsers || [];
    98→  });
    99→
   100→  // Quick Action triggers removed per user request — chat input is now clean.
   101→
   102→  // Fetch messages of specified subject channel.
   103→  // If `since` is provided, uses long-polling (server holds up to 20s waiting
   104→  // for new messages). Otherwise, fetches the latest 200 messages immediately.
   105→  const fetchChannelMessages = async (subjId: string, silent: boolean = false, since?: string) => {
   106→    if (!silent) {
   107→      setLoading(true);
   108→      setErrorMessage(null);
   109→    }
   110→    try {
   111→      // Always use /api/chat/[subjectId] — 'general' maps to global (null subjectId)
   112→      const endpoint = `/api/chat/${subjId}` + (since ? `?since=${encodeURIComponent(since)}` : '');
   113→      const res = await apiFetch(endpoint);
   114→      if (res.ok) {
   115→        const raw = await res.json();
   116→        // Response shape: { messages: [...], polledAt: '...' }
   117→        // Handle both old (array) and new ({messages}) shapes for resilience
   118→        const rawMessages = Array.isArray(raw) ? raw : (raw.messages || []);
   119→        // Normalize API fields (text -> content, userAvatar -> userProfilePic) to match local MessageType shape
   120→        const data: MessageType[] = rawMessages.map((m: any) => ({
   121→          id: m.id,
   122→          _id: m._id,
   123→          userId: m.userId,
   124→          userName: m.userName,
   125→          userRole: m.userRole === 'admin' ? 'admin' : 'user',
   126→          content: m.text ?? m.content ?? '',
   127→          subjectId: m.subjectId ?? subjId,
   128→          createdAt: m.createdAt,
   129→          userProfilePic: m.userProfilePic ?? m.userAvatar,
   130→          imageUrl: m.imageUrl,
   131→        }));
   132→        setMessages((prev) => {
   133→          if (since) {
   134→            // Long-poll mode — merge new messages into existing
   135→            if (data.length === 0) return prev;
   136→            const existingIds = new Set(prev.map(m => m.id || m._id));
   137→            const newOnes = data.filter(m => !existingIds.has(m.id || m._id));
   138→            return newOnes.length > 0 ? [...prev, ...newOnes] : prev;
   139→          }
   140→          // Initial fetch — replace all
   141→          return data;
   142→        });
   143→      } else if (!silent) {
   144→        const errData = await res.json();
   145→        setErrorMessage(errData.error || t('networkLoss'));
   146→      }
   147→    } catch (e: any) {
   148→      if (!silent) {
   149→        setErrorMessage(t('networkLoss'));
   150→      }
   151→    } finally {
   152→      if (!silent) {
   153→        setLoading(false);
   154→      }
   155→    }
   156→  };
   157→
   158→  // Image picker — converts to base64 data URL for inline upload
   159→  const handleImagePick = (e: React.ChangeEvent<HTMLInputElement>) => {
   160→    const file = e.target.files?.[0];
   161→    if (imageInputRef.current) imageInputRef.current.value = '';
   162→    if (!file) return;
   163→    if (!file.type.startsWith('image/')) {
   164→      setErrorMessage('Please choose an image file.');
   165→      return;
   166→    }
   167→    if (file.size > 2 * 1024 * 1024) {
   168→      setErrorMessage('Max 2 MB for chat images.');
   169→      return;
   170→    }
   171→    const reader = new FileReader();
   172→    reader.onload = () => setPendingImage(String(reader.result || ''));
   173→    reader.onerror = () => setErrorMessage('Could not read image file.');
   174→    reader.readAsDataURL(file);
   175→  };
   176→
   177→  // Perform post of typed message content + optional image
   178→  const handleSendMessage = async (incomingContent?: string) => {
   179→    const targetContent = incomingContent || typedMessage;
   180→    // Need either text or an image to send
   181→    if ((!targetContent.trim() && !pendingImage) || submitting) return;
   182→
   183→    setSubmitting(true);
   184→    setErrorMessage(null);
   185→
   186→    const messageContent = targetContent;
   187→    const imageToSend = pendingImage;
   188→    if (!incomingContent) {
   189→      setTypedMessage('');
   190→    }
   191→    setPendingImage(null);
   192→
   193→    try {
   194→      // Next.js /api/chat expects { text, subjectId, imageUrl } — pass null for general lounge
   195→      const subjectPayload = activeChannelId === 'general' ? null : activeChannelId;
   196→      const res = await apiFetch('/api/chat', {
   197→        method: 'POST',
   198→        headers: {
   199→          'Content-Type': 'application/json',
   200→        },
   201→        body: JSON.stringify({
   202→          text: messageContent || '(image)',
   203→          subjectId: subjectPayload,
   204→          imageUrl: imageToSend || undefined,
   205→        })
   206→      });
   207→
   208→      if (res.ok) {
   209→        const newMsg = await res.json();
   210→        // Normalize the new message to match local MessageType shape
   211→        const normalized: MessageType = {
   212→          id: newMsg.id,
   213→          _id: newMsg._id,
   214→          userId: newMsg.userId,
   215→          userName: newMsg.userName,
   216→          userRole: newMsg.userRole === 'admin' ? 'admin' : 'user',
   217→          content: newMsg.text ?? newMsg.content ?? messageContent,
   218→          subjectId: newMsg.subjectId ?? activeChannelId,
   219→          createdAt: newMsg.createdAt,
   220→          userProfilePic: newMsg.userProfilePic ?? newMsg.userAvatar,
   221→          imageUrl: newMsg.imageUrl,
   222→        };
   223→        setMessages((prev) => [...prev, normalized]);
   224→        // Force scroll-to-bottom on newly typed posts from our side
   225→        setTimeout(() => {
   226→          if (scrollContainerRef.current) {
   227→            scrollContainerRef.current.scrollTop = scrollContainerRef.current.scrollHeight;
   228→          }
   229→        }, 80);
   230→      } else {
   231→        const errData = await res.json();
   232→        setErrorMessage(errData.error || "Failed to broadcast message.");
   233→        if (!incomingContent) setTypedMessage(messageContent); // Restore text on failure
   234→      }
   235→    } catch (err) {
   236→      setErrorMessage("Broadcast failed due to structural network discrepancy.");
   237→      if (!incomingContent) setTypedMessage(messageContent); // Restore text on failure
   238→    } finally {
   239→      setSubmitting(false);
   240→    }
   241→  };
   242→
   243→  const handleFormSubmit = async (e: React.FormEvent) => {
   244→    e.preventDefault();
   245→    await handleSendMessage();
   246→  };
   247→
   248→  // Perform message eradication (admin or owner only)
   249→  const handleDeleteMessage = async (msgId: string) => {
   250→    if (!msgId) return;
   251→
   252→    try {
   253→      // Next.js /api/chat delete via ?id= query param
   254→      const res = await apiFetch(`/api/chat?id=${msgId}`, {
   255→        method: 'DELETE',
   256→      });
   257→
   258→      if (res.ok) {
   259→        // Remove locally from UI state
   260→        setMessages((prev) => prev.filter(m => (m.id || m._id) !== msgId));
   261→      } else {
   262→        const errData = await res.json();
   263→        setErrorMessage(errData.error || "Could not delete message."); setTimeout(() => setErrorMessage(null), 4000);
   264→      }
   265→    } catch (err) {
   266→      setErrorMessage("Network error deleting message."); setTimeout(() => setErrorMessage(null), 4000);
   267→    }
   268→  };
   269→
   270→  // ── Client-side polling (Vercel serverless-safe) ──────────────────────
   271→  // Instead of long-polling (server holds request open 20s — burns Vercel
   272→  // quota + times out on Hobby plan), the client polls every 3 seconds.
   273→  // The server returns immediately with any new messages since the last
   274→  // poll. This is cheaper, simpler, and works on all serverless platforms.
   275→  useEffect(() => {
   276→    let cancelled = false;
   277→
   278→    const startPollLoop = async (subjId: string) => {
   279→      // Initial fetch (no `since` param — get latest 200 messages)
   280→      await fetchChannelMessages(subjId, false);
   281→
   282→      if (cancelled) return;
   283→
   284→      // Poll loop — fetch new messages every 3 seconds
   285→      while (!cancelled) {
   286→        await new Promise((r) => setTimeout(r, 3000));
   287→        if (cancelled) break;
   288→
   289→        // Get the latest message timestamp to use as the `since` cursor
   290→        const latestMsg = messagesRef.current[messagesRef.current.length - 1];
   291→        const since = latestMsg?.createdAt || new Date(0).toISOString();
   292→
   293→        // Fetch only new messages since the last poll
   294→        await fetchChannelMessages(subjId, true, since);
   295→      }
   296→    };
   297→
   298→    startPollLoop(activeChannelId);
   299→
   300→    return () => {
   301→      cancelled = true;
   302→    };
   303→  }, [activeChannelId]);
   304→
   305→  // Listen to WebSocket-triggered immediate reload event for instantaneous responsiveness
   306→  useEffect(() => {
   307→    const handleRemoteChatUpdate = (e: Event) => {
   308→      const customEvent = e as CustomEvent;
   309→      if (customEvent.detail && customEvent.detail.subjectId === activeChannelId) {
   310→        fetchChannelMessages(activeChannelId, true);
   311→      }
   312→    };
   313→
   314→    window.addEventListener('chat-updated', handleRemoteChatUpdate);
   315→    return () => {
   316→      window.removeEventListener('chat-updated', handleRemoteChatUpdate);
   317→    };
   318→  }, [activeChannelId]);
   319→
   320→  // Sync initial components presence data and update when event triggers
   321→  useEffect(() => {
   322→    if ((window as any).__activeUsers) {
   323→      setActiveUsers((window as any).__activeUsers);
   324→    }
   325→    const handlePresenceUpdated = (e: Event) => {
   326→      const customEvent = e as CustomEvent;
   327→      if (customEvent.detail && customEvent.detail.activeUsers) {
   328→        setActiveUsers(customEvent.detail.activeUsers);
   329→      }
   330→    };
   331→    window.addEventListener('presence-updated', handlePresenceUpdated);
   332→    return () => {
   333→      window.removeEventListener('presence-updated', handlePresenceUpdated);
   334→    };
   335→  }, []);
   336→
   337→  // Handle user-friendly quiet auto-scrolling on load or when sending message
   338→  const lastMessageCountRef = useRef<number>(0);
   339→  useEffect(() => {
   340→    const scrollContainer = scrollContainerRef.current;
   341→    if (!scrollContainer) return;
   342→
   343→    if (messages.length > lastMessageCountRef.current) {
   344→      const lastMsg = messages[messages.length - 1];
   345→      const isMyMessage = lastMsg && lastMsg.userId === user?.id;
   346→      const isInitialScroll = !loading && lastMessageCountRef.current === 0;
   347→
   348→      // Scroll to bottom immediately if it's the initial load or if the user sent a new message
   349→      if (isInitialScroll || isMyMessage) {
   350→        scrollContainer.scrollTop = scrollContainer.scrollHeight;
   351→      } else {
   352→        // If someone else posted, scroll down only if the user is already near the bottom
   353→        // (to prevent interrupting their reading of history)
   354→        const threshold = 180; // pixels from bottom threshold zone
   355→        const isNearBottom = scrollContainer.scrollHeight - scrollContainer.scrollTop - scrollContainer.clientHeight < threshold;
   356→        if (isNearBottom) {
   357→          scrollContainer.scrollTop = scrollContainer.scrollHeight;
   358→        }
   359→      }
   360→    }
   361→    lastMessageCountRef.current = messages.length;
   362→  }, [messages, loading, user?.id]);
   363→
   364→  const activeChannelObj = subjects.find(s => s.id === activeChannelId);
   365→  const activeChannelTitle = activeChannelId === 'general' ? 'General Academic Lounge' : `${activeChannelObj?.title} Support Channel`;
   366→  const activeChannelDesc = activeChannelId === 'general'
   367→    ? 'Universal classroom bulletin board and loose peer conversation space.'
   368→    : activeChannelObj?.description;
   369→
   370→  const formatDateLabel = (isoStr: string) => {
   371→    try {
   372→      const date = new Date(isoStr);
   373→      return date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
   374→    } catch (e) {
   375→      return '';
   376→    }
   377→  };
   378→
   379→  // Assign specific visual descriptors, emojis, and icons of subjects for professional feel
   380→  const getSubjectIcon = (title: string, sizeClass = "w-4 h-4") => {
   381→    const l = title.toLowerCase();
   382→    if (l.includes('physic')) return <Compass className={`${sizeClass} text-indigo-400`} />;
   383→    if (l.includes('chemist')) return <Beaker className={`${sizeClass} text-emerald-400`} />;
   384→    if (l.includes('biolog')) return <Sparkle className={`${sizeClass} text-amber-400`} />;
   385→    if (l.includes('math')) return <GraduationCap className={`${sizeClass} text-pink-400`} />;
   386→    if (l.includes('ict')) return <Code className={`${sizeClass} text-cyan-400`} />;
   387→    return <Hash className={`${sizeClass} text-slate-450`} />;
   388→  };
   389→
   390→  const getSubjectGradient = (title: string) => {
   391→    const l = title.toLowerCase();
   392→    if (l.includes('physic')) return 'from-indigo-505/20 to-indigo-950/25 border-indigo-500/20 text-indigo-350';
   393→    if (l.includes('chemist')) return 'from-emerald-505/20 to-emerald-950/25 border-emerald-500/20 text-emerald-350';
   394→    if (l.includes('biolog')) return 'from-amber-505/20 to-amber-950/25 border-amber-500/20 text-amber-350';
   395→    if (l.includes('math')) return 'from-pink-505/20 to-pink-950/25 border-pink-500/20 text-pink-350';
   396→    if (l.includes('ict')) return 'from-cyan-505/20 to-cyan-950/25 border-cyan-500/20 text-cyan-350';
   397→    return 'from-slate-800/10 to-slate-950/20 border-white/5 text-slate-350';
   398→  };
   399→
   400→  return (
   401→    <div
   402→      id="classroom-chat-root"
   403→      className="
   404→        w-full min-h-[600px] lg:h-[calc(100vh-180px)]
   405→        flex flex-col lg:flex-row
   406→        rounded-3xl border border-white/[0.06]
   407→        bg-slate-950/40 overflow-hidden
   408→        backdrop-blur-2xl relative shadow-2xl
   409→      "
   410→    >
   411→      {/* ───────────────────────────────────────────────────────────────────
   412→          CHANNELS SIDEBAR (left on desktop, full-screen on mobile when mobileView='channels')
   413→          ─────────────────────────────────────────────────────────────────── */}
   414→      <aside
   415→        className={`
   416→          w-full lg:w-80 xl:w-96 shrink-0
   417→          flex flex-col gap-3
   418→          bg-slate-950/70 border-r border-white/[0.04]
   419→          overflow-hidden
   420→          ${mobileView === 'channels' ? 'flex' : 'hidden lg:flex'}
   421→        `}
   422→      >
   423→        {/* Sidebar header */}
   424→        <div className="flex items-center justify-between px-4 py-3 border-b border-white/[0.04] shrink-0">
   425→          <div className="flex items-center gap-2 min-w-0">
   426→            <div className="p-1.5 rounded-lg bg-pink-500/10 border border-pink-500/20 shrink-0">
   427→              <Users className="w-3.5 h-3.5 text-pink-400" />
   428→            </div>
   429→            <span className="font-mono text-[10px] uppercase font-black tracking-widest text-slate-300 truncate">
   430→              {t('lobby')}
   431→            </span>
   432→          </div>
   433→          <span className="flex items-center gap-1 text-[9px] bg-emerald-500/10 text-emerald-400 border border-emerald-500/20 px-1.5 py-0.5 rounded font-mono font-bold">
   434→            <span className="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-pulse" />
   435→            {t('synced')}
   436→          </span>
   437→        </div>
   438→
   439→        {/* Active users strip */}
   440→        <div className="px-3 shrink-0">
   441→          <div className="bg-slate-900/40 border border-white/[0.03] p-2.5 rounded-2xl">
   442→            <div className="flex items-center justify-between mb-2">
   443→              <span className="text-[9px] font-bold text-slate-500 uppercase tracking-wider">{t('activeScholars')}</span>
   444→              <span className="text-[9px] font-mono text-cyan-400">{activeUsers.length} {t('onlineSuffix')}</span>
   445→            </div>
   446→            <div className="flex items-center gap-2.5 overflow-x-auto pb-1 scrollbar-none">
   447→              {activeUsers.length === 0 ? (
   448→                <span className="text-[10px] text-slate-500 italic px-1">Connecting…</span>
   449→              ) : (
   450→                activeUsers.slice(0, 8).map((item, idx) => (
   451→                  <div key={item.id || idx} className="relative group shrink-0 flex flex-col items-center gap-1" title={`${item.name} (${item.role})`}>
   452→                    <div className="relative">
   453→                      {item.profilePic ? (
   454→                        <img
   455→                          src={item.profilePic}
   456→                          alt={item.name}
   457→                          referrerPolicy="no-referrer"
   458→                          className="w-8 h-8 rounded-full object-cover border border-emerald-500/60 bg-slate-950"
   459→                        />
   460→                      ) : (
   461→                        <div className="w-8 h-8 rounded-full border border-emerald-500/60 bg-slate-900 flex items-center justify-center text-slate-200 text-[10px] font-bold">
   462→                          {item.name ? item.name.charAt(0).toUpperCase() : '?'}
   463→                        </div>
   464→                      )}
   465→                      <span className="absolute -bottom-0.5 -right-0.5 w-2.5 h-2.5 rounded-full bg-emerald-500 border-2 border-slate-950" />
   466→                    </div>
   467→                    <span className="text-[8px] text-slate-400 font-bold max-w-[44px] truncate leading-none text-center">
   468→                      {item.name ? item.name.split(' ')[0] : 'Scholar'}
   469→                    </span>
   470→                  </div>
   471→                ))
   472→              )}
   473→            </div>
   474→          </div>
   475→        </div>
   476→
   477→        {/* Channel list */}
   478→        <div className="flex-1 overflow-y-auto px-3 pb-3 space-y-1.5 scrollbar-thin">
   479→          <span className="block text-[9px] font-bold text-slate-500 uppercase tracking-widest px-2 pb-1 font-mono">
   480→            {t('chatRooms')}
   481→          </span>
   482→
   483→          {/* General channel */}
   484→          <button
   485→            onClick={() => {
   486→              setActiveChannelId('general');
   487→              setMobileView('chat');
   488→            }}
   489→            className={`w-full flex items-center gap-3 px-3 py-2.5 min-h-[48px] rounded-2xl text-xs font-semibold cursor-pointer transition-all border text-left ${
   490→              activeChannelId === 'general'
   491→                ? 'bg-gradient-to-r from-cyan-500/15 to-indigo-500/10 border-cyan-500/30 text-cyan-200'
   492→                : 'text-slate-400 hover:text-slate-200 hover:bg-white/[0.03] border-transparent'
   493→            }`}
   494→          >
   495→            <div className="p-2 rounded-xl bg-slate-800/60 border border-white/5 shrink-0">
   496→              <Hash className="w-3.5 h-3.5" />
   497→            </div>
   498→            <div className="min-w-0 flex-1">
   499→              <div className="font-bold truncate">General Lounge</div>
   500→              <div className="text-[9px] text-slate-500 truncate">Open discussion for everyone</div>
   501→            </div>
   502→          </button>
   503→
   504→          {/* Subject channels */}
   505→          {subjects.map((subj) => {
   506→            const isActive = activeChannelId === subj.id;
   507→            return (
   508→              <button
   509→                key={subj.id}
   510→                onClick={() => {
   511→                  setActiveChannelId(subj.id);
   512→                  setMobileView('chat');
   513→                }}
   514→                className={`w-full flex items-center gap-3 px-3 py-2.5 min-h-[48px] rounded-2xl text-xs font-semibold cursor-pointer transition-all border text-left ${
   515→                  isActive
   516→                    ? 'bg-gradient-to-r from-cyan-500/15 to-indigo-500/10 border-cyan-500/30 text-cyan-200'
