   134→/* ---------------- Fourier series constants ---------------- */

   136→// Square wave Fourier series:  f(t) = (4/π) · Σ_{k=1..7} sin((2k-1)·ω·t) / (2k−1)
   137→// 7 harmonics — odd multiples of the base frequency (1, 3, 5, 7, 9, 11, 13).
   138→const HARMONIC_KS = [1, 3, 5, 7, 9, 11, 13];
   139→const HARMONIC_AMPS = HARMONIC_KS.map((k) => (4 / Math.PI) / k);

   141→const SUBSCRIPTS: Record<string, string> = {
   142→  '0': '₀', '1': '₁', '2': '₂', '3': '₃', '4': '₄',
   143→  '5': '₅', '6': '₆', '7': '₇', '8': '₈', '9': '₉',
   144→};

   146→const toSubscript = (n: number): string =>
   147→  String(n).split('').map((c) => SUBSCRIPTS[c] ?? c).join('');

   166→/* ---------------- AnimatedStat (kept — one-shot rAF count-up) ---------------- */
   167→
   168→const AnimatedStat: React.FC<{ value: number; className?: string }> = ({ value, className = '' }) => {
   169→  const ref = useRef<HTMLSpanElement>(null);
   170→  const inView = useInView(ref, { once: true, margin: '-40px' });
   171→  const [display, setDisplay] = useState(0);
   172→
   173→  useEffect(() => {
   174→    if (!inView) return;
   175→    const prefersReducedMotion =
   176→      typeof window !== 'undefined' && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
   177→    if (prefersReducedMotion) {
   178→      setDisplay(value);
   179→      return;
   180→    }
   181→    let raf = 0;
   182→    const start = performance.now();
   183→    const duration = 1400;
   184→    const tick = (now: number) => {
   185→      const t = Math.min((now - start) / duration, 1);
   186→      const eased = 1 - Math.pow(1 - t, 3);
   187→      setDisplay(Math.round(eased * value));
   188→      if (t < 1) raf = requestAnimationFrame(tick);
   189→    };
   190→    raf = requestAnimationFrame(tick);
   191→    return () => cancelAnimationFrame(raf);
   192→  }, [inView, value]);
   193→
   194→  return <span ref={ref} className={className}>{display}</span>;
   195→};

   197→/* ---------------- ParallaxSectionHeading (desktop only — GSAP scroll-linked transform, no framer-motion) ---------------- */
   198→
   199→const ParallaxSectionHeading: React.FC<{ children: React.ReactNode; className?: string }> = ({
   200→  children,
   201→  className,
   202→}) => {
   203→  const ref = useRef<HTMLDivElement>(null);
   204→  // Only apply parallax on desktop — on mobile, scroll listeners + transforms cause scroll jank
   205→  const [isDesktop, setIsDesktop] = useState(false);
   206→  useEffect(() => {
   207→    const check = () => setIsDesktop(window.innerWidth >= 768);
   208→    check();
   209→    window.addEventListener('resize', check);
   210→    return () => window.removeEventListener('resize', check);
   211→  }, []);
   212→  useEffect(() => {
   213→    if (!isDesktop) return;
   214→    const el = ref.current;
   215→    if (!el) return;
   216→    let raf = 0;
   217→    const update = () => {
   218→      const rect = el.getBoundingClientRect();
   219→      const viewportH = window.innerHeight;
   220→      const center = rect.top + rect.height / 2 - viewportH / 2;
   221→      const progress = Math.max(-1, Math.min(1, center / viewportH));
   222→      el.style.transform = `translateY(${-progress * 40}px)`;
   223→    };
   224→    const onScroll = () => {
   225→      cancelAnimationFrame(raf);
   226→      raf = requestAnimationFrame(update);
   227→    };
   228→    update();
   229→    window.addEventListener('scroll', onScroll, { passive: true });
   230→    return () => {
   231→      window.removeEventListener('scroll', onScroll);
   232→      cancelAnimationFrame(raf);
   233→    };
   234→  }, [isDesktop]);
   235→  return (
   236→    <div ref={ref} className={className}>
   237→      {children}
   238→    </div>
   239→  );
   240→};

   242→/* ---------------- GsapHeading (SplitText char stagger on scroll-in — one-shot) ---------------- */
   243→
   244→const GsapHeading: React.FC<{
   245→  as: 'h1' | 'h2';
   246→  children: React.ReactNode;
   247→  className?: string;
   248→}> = ({ as, children, className }) => {
   249→  const ref = useRef<HTMLHeadingElement>(null);
   250→  const splitRef = useRef<SplitText | null>(null);
   251→  useEffect(() => {
   252→    const el = ref.current;
   253→    if (!el) return;
   254→    gsap.registerPlugin(SplitText);
   255→
   256→    // Ensure the heading is visible immediately (in case the IntersectionObserver
   257→    // doesn't fire on mount — e.g. element is already in viewport on load).
   258→    gsap.set(el, { opacity: 1 });
   259→
   260→    const observer = new IntersectionObserver(
   261→      (entries) => {
   262→        entries.forEach((entry) => {
   263→          if (entry.isIntersecting) {
   264→            const split = new SplitText(el, { type: 'chars,words' });
   265→            splitRef.current = split;
   266→            // Character-by-character reveal: each char flies in from below
   267→            // with a slight rotation + scale for a premium staggered effect.
   268→            gsap.from(split.chars, {
   269→              opacity: 0,
   270→              y: 20,
   271→              rotationX: -90,
   272→              scale: 0.5,
   273→              duration: 0.6,
   274→              stagger: 0.025,
   275→              ease: 'back.out(1.7)',
   276→              transformOrigin: '50% 100%',
   277→            });
   278→            observer.unobserve(el);
   279→          }
   280→        });
   281→      },
   282→      { rootMargin: '-40px' },
   283→    );
   284→    observer.observe(el);
   285→    return () => {
   286→      observer.disconnect();
   287→      if (splitRef.current) splitRef.current.revert();
   288→    };
   289→  }, []);
   290→  const Tag = as;
   291→  return (
   292→    <Tag ref={ref} className={className} style={{ opacity: 1 }}>
   293→      {children}
   294→    </Tag>
   295→  );
   296→};

   318→  const lineRefs = useRef<Array<SVGLineElement | null>>(Array(7).fill(null));
   319→  const tipRef = useRef<SVGCircleElement>(null);

   323→  const sumLabelRef = useRef<HTMLSpanElement>(null);

   325→  const chipRef = useRef<HTMLDivElement>(null);

   327→  /* CTA dots + glow animation refs */
   328→  const ctaDotsRef = useRef<HTMLDivElement>(null);
   329→  const ctaGlow1Ref = useRef<HTMLDivElement>(null);
   330→  const ctaGlow2Ref = useRef<HTMLDivElement>(null);
   331→
   332→  /* Parallax transforms — only on desktop (≥768px) to prevent mobile scroll jank.
   333→     On mobile, will-change:transform + useScroll listeners cause the scroll to
   334→     stutter badly. We detect the viewport and conditionally apply parallax. */
   335→  const [isDesktop, setIsDesktop] = useState(false);
   336→  useEffect(() => {
   337→    const check = () => setIsDesktop(window.innerWidth >= 768);
   338→    check();
   339→    window.addEventListener('resize', check);
   340→    return () => window.removeEventListener('resize', check);
   341→  }, []);
   342→  const { scrollY } = useScroll();
   343→  // Spring-smoothed hero parallax — useSpring wraps the raw scroll transform
   344→  // so fast scrolling doesn't feel jerky. The spring gently catches up to
   345→  // the scroll position instead of snapping instantly.
   346→  // NOTE: useTransform with a function (not static arrays) so the output
   347→  // range re-evaluates when isDesktop changes. Static arrays capture the
   348→  // value of isDesktop at hook-call time and never update.
   349→  const rawHeroTextY = useTransform(scrollY, (v) => isDesktop ? v * -0.3 : v * -0.12);
   350→  const rawHeroVizY = useTransform(scrollY, (v) => isDesktop ? v * -0.5 : v * -0.2);

   352→  const heroVizY = useSpring(rawHeroVizY, { stiffness: 100, damping: 30, mass: 0.5 });
   353→

   355→     Each moves at a different speed to create depth.
   356→     Mobile uses 40% of the desktop effect — subtle but visible. */
   357→  const rawBgFarY = useTransform(scrollY, (v) => isDesktop ? v * 0.0875 : v * 0.035);
   358→  const rawBgMidY = useTransform(scrollY, (v) => isDesktop ? v * 0.125 : v * 0.05);
   359→  const rawBgNearY = useTransform(scrollY, (v) => isDesktop ? v * 0.175 : v * 0.07);
   360→  const bgFarMountainY = useSpring(rawBgFarY, { stiffness: 80, damping: 25, mass: 0.8 });
   361→  const bgMidMountainY = useSpring(rawBgMidY, { stiffness: 80, damping: 25, mass: 0.8 });
   362→  const bgNearHillY = useSpring(rawBgNearY, { stiffness: 80, damping: 25, mass: 0.8 });
   363→
   364→  /* ── Content section parallax ──
   365→     Each content section gets a subtle vertical parallax offset so the
   366→     whole page feels 3D — not just the hero. Sections alternate between
   367→     "near" (move up faster) and "far" (move up slower) to create depth.
   368→     Mobile uses 40% of the desktop effect. */
   369→  const sectionY1 = useTransform(scrollY, (v) => isDesktop ? v * -0.02 : v * -0.008);
   370→  const sectionY2 = useTransform(scrollY, (v) => isDesktop ? v * -0.04 : v * -0.016);
   371→  const sectionY3 = useTransform(scrollY, (v) => isDesktop ? v * -0.03 : v * -0.012);
   372→  const sectionY4 = useTransform(scrollY, (v) => isDesktop ? v * -0.05 : v * -0.02);
   373→  const sectionY5 = useTransform(scrollY, (v) => isDesktop ? v * -0.025 : v * -0.01);
   374→  const sectionY6 = useTransform(scrollY, (v) => isDesktop ? v * -0.06 : v * -0.024);
   375→
   376→  /* ── State-of-the-art mouse parallax ──
   377→     The cursor position drives a spring-smoothed motion value that's mapped
   378→     to horizontal + vertical offsets for each star layer at different rates
   379→     (distant slow, near fast). Combined with the scroll-Y parallax above,
   380→     this creates a buttery 3D depth effect as the user moves their mouse.
   381→     Uses requestAnimationFrame throttling to avoid scroll jank. */
   382→  const mouseX = useMotionValue(0);
   383→  const springMouseX = useSpring(mouseX, { stiffness: 60, damping: 20, mass: 0.6 });
   384→  // Distant layer — moves slowest (0.3x of mouse delta)
   385→  const farMouseX = useTransform(springMouseX, [-0.5, 0.5], [-10, 10]);
   386→  // Mid layer — medium (0.7x)
   387→  const midMouseX = useTransform(springMouseX, [-0.5, 0.5], [-25, 25]);

   389→  const nearMouseX = useTransform(springMouseX, [-0.5, 0.5], [-40, 40]);
   390→
   391→  // Throttled mouse handler — uses rAF to avoid flooding the main thread.
   392→  const rafRef = useRef<number | null>(null);
   393→  const handleMouseParallax = (e: React.MouseEvent) => {
   394→    if (rafRef.current !== null) return;
   395→    rafRef.current = requestAnimationFrame(() => {
   396→      rafRef.current = null;
   397→      const x = e.clientX / window.innerWidth - 0.5;
   398→      mouseX.set(x);
   399→    });
   400→  };
   401→  const resetMouseParallax = () => {
   402→    if (rafRef.current !== null) {
   403→      cancelAnimationFrame(rafRef.current);
   404→      rafRef.current = null;
   405→    }
   406→    mouseX.set(0);
   407→  };

   417→  }, [mobileNavOpen]);
   418→
   419→  /* GSAP-animated Fourier series epicycles — infinite rotation.
   420→     NOTE: the floating math glyphs (∑ π ω ƒ) were removed per user request
   421→     (they looked like "floating balls"). Only the epicycles + waveform remain. */
   422→  useEffect(() => {
   423→    if (typeof window === 'undefined') return;
   424→    const prefersReduced =
   425→      typeof window !== 'undefined' &&
   426→      window.matchMedia('(prefers-reduced-motion: reduce)').matches;
   427→
   428→    const tweens: gsap.core.Tween[] = [];
   429→    const timelines: gsap.core.Timeline[] = [];
   430→
   431→    const FOURIER_SCALE = 36;
   432→    const FOURIER_OMEGA_BASE = 1.4;
   433→    const WAVE_SAMPLES = 120;
   434→    const WAVE_WINDOW = 6;
   435→    const WAVE_X_START = 120;
   436→    const WAVE_X_END = 240;
   437→    const WAVE_X_RANGE = WAVE_X_END - WAVE_X_START;
   438→    const EPICYCLE_CENTER_X = 60;
   439→    const EPICYCLE_CENTER_Y = 80;
   440→
   441→    const omegas = HARMONIC_KS.map((k) => k * FOURIER_OMEGA_BASE);
   442→
   443→    function fourierSum(s: number): number {
   444→      let sum = 0;
   445→      for (let i = 0; i < HARMONIC_KS.length; i++) {
   446→        sum += HARMONIC_AMPS[i] * Math.sin(omegas[i] * s);
   447→      }
   448→      return sum;
   449→    }
   450→
   451→    let t = 0;
   452→    let lastFrameTime = performance.now();
   453→
   454→    function renderFrame(currentTime: number) {
   455→      const dt = Math.min((currentTime - lastFrameTime) / 1000, 0.05);
   456→      lastFrameTime = currentTime;
   457→      t += dt;
   458→
   459→      let prevX = EPICYCLE_CENTER_X;
   460→      let prevY = EPICYCLE_CENTER_Y;
   461→      let sumValue = 0;
   462→
   463→      for (let i = 0; i < HARMONIC_KS.length; i++) {
   464→        const angle = omegas[i] * t;
   465→        const dx = HARMONIC_AMPS[i] * FOURIER_SCALE * Math.cos(angle);
   466→        const dy = -HARMONIC_AMPS[i] * FOURIER_SCALE * Math.sin(angle);
   467→        const tipX = prevX + dx;
   468→        const tipY = prevY + dy;
   469→
   470→        const circleEl = circleRefs.current[i];
   471→        if (circleEl) {
   472→          circleEl.setAttribute('cx', prevX.toFixed(2));
   473→          circleEl.setAttribute('cy', prevY.toFixed(2));
   474→        }
   475→        const lineEl = lineRefs.current[i];
   476→        if (lineEl) {
   477→          lineEl.setAttribute('x1', prevX.toFixed(2));
   478→          lineEl.setAttribute('y1', prevY.toFixed(2));
   479→          lineEl.setAttribute('x2', tipX.toFixed(2));
   480→          lineEl.setAttribute('y2', tipY.toFixed(2));
   481→        }
   482→
   483→        prevX = tipX;
   484→        prevY = tipY;
   485→        sumValue += HARMONIC_AMPS[i] * Math.sin(omegas[i] * t);
   486→      }
   487→
   488→      if (tipRef.current) {
   489→        tipRef.current.setAttribute('cx', prevX.toFixed(2));
   490→        tipRef.current.setAttribute('cy', prevY.toFixed(2));
   491→      }
   492→
   493→      if (waveformRef.current) {
   494→        let points = '';
