   133→   133→/* ---------------- Fourier series constants ---------------- */
   134→   134→
   135→   135→// Square wave Fourier series:  f(t) = (4/π) · Σ_{k=1..7} sin((2k-1)·ω·t) / (2k-1)
   136→   136→// 7 harmonics — odd multiples of the base frequency (1, 3, 5, 7, 9, 11, 13).
   137→   137→const HARMONIC_KS = [1, 3, 5, 7, 9, 11, 13];
   138→   138→const HARMONIC_AMPS = HARMONIC_KS.map((k) => (4 / Math.PI) / k);

   140→   140→const SUBSCRIPTS: Record<string, string> = {

   143→   143→};
   144→   144→
   145→   145→const toSubscript = (n: number): string =>

   173→   173→/* ---------------- AnimatedStat (kept — one-shot rAF count-up) ---------------- */
   174→   174→
   175→   175→const AnimatedStat: React.FC<{ value: number; className?: string }> = ({ value, className = '' }) => {
   176→   176→  const ref = useRef<HTMLSpanElement>(null);
   177→   177→  const inView = useInView(ref, { once: true, margin: '-40px' });
   178→   178→  const [display, setDisplay] = useState(0);
   179→   179→
   180→   180→  useEffect(() => {
   181→   181→    if (!inView) return;
   182→   182→    const prefersReducedMotion =
   183→   183→      typeof window !== 'undefined' && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
   184→   184→    if (prefersReducedMotion) {
   185→   185→      setDisplay(value);
   186→   186→      return;
   187→   187→    }
   188→   188→    let raf = 0;
   189→   189→    const start = performance.now();
   190→   190→    const duration = 1400;
   191→   191→    const tick = (now: number) => {
   192→   192→      const t = Math.min((now - start) / duration, 1);
   193→   193→      const eased = 1 - Math.pow(1 - t, 3);
   194→   194→      setDisplay(Math.round(eased * value));
   195→   195→      if (t < 1) raf = requestAnimationFrame(tick);
   196→   196→    };
   197→   197→    raf = requestAnimationFrame(tick);
   198→   198→    return () => cancelAnimationFrame(raf);
   199→   199→  }, [inView, value]);
   200→   200→
   201→   201→  return <span ref={ref} className={className}>{display}</span>;
   202→   202→};

   215→   215→  useEffect(() => {
   216→   216→    const check = () => setIsDesktop(window.innerWidth >= 768);
   217→   217→    check();
   218→   218→    window.addEventListener('resize', check);
   219→   219→    return () => window.removeEventListener('resize', check);
   220→   220→  }, []);
   221→   221→  return (
   222→   222→    <motion.div ref={ref} style={{ y: isDesktop ? y : 0, willChange: isDesktop ? 'transform' as const : undefined }} className={className}>
   223→   223→      {children}
   224→   224→    </motion.div>

   229→   229→
   230→   230→export const LandingPage: React.FC<LandingPageProps> = ({
   231→   231→  onEnter,
   232→   232→  isAuthenticated,
   233→   233→  onGoToDashboard,
   234→   234→  subjects = [],
   235→   235→  bannerConfig,
   236→   236→  folders = [],
   237→   237→  announcements = [],
   238→   238→  onRegister,
   239→   239→  onSignIn,
   240→   240→  onBrowseMarketplace,
   241→   241→}) => {
   242→   242→  const [mobileNavOpen, setMobileNavOpen] = useState(false);
   243→   243→  const [openFaq, setOpenFaq] = useState<number | null>(0);
   244→   244→
   245→   245→  /* GSAP refs for the Fourier series epicycle animation */
   246→   246→  const sceneRef = useRef<HTMLDivElement>(null);
   247→   247→  const circleRefs = useRef<Array<SVGCircleElement | null>>(Array(7).fill(null));
   248→   248→  const lineRefs = useRef<Array<SVGLineElement | null>>(Array(7).fill(null));
   249→   249→  const tipRef = useRef<SVGCircleElement>(null);
   250→   250→  const trailLineRef = useRef<SVGLineElement>(null);
   251→   251→  const waveformRef = useRef<SVGPolylineElement>(null);
   252→   252→  const tLabelRef = useRef<HTMLSpanElement>(null);
   253→   253→  const sumLabelRef = useRef<HTMLSpanElement>(null);
   254→   254→  const stepRowsRef = useRef<HTMLDivElement>(null);
   255→   255→  const chipRef = useRef<HTMLDivElement>(null);
   256→   256→  const glyphRefs = useRef<Array<HTMLSpanElement | null>>([null, null, null, null]);
   257→   257→
   258→   258→  /* Parallax transforms — only on desktop (≥768px) to prevent mobile scroll jank.
   259→   259→     On mobile, will-change:transform + useScroll listeners cause the scroll to
   260→   260→     stutter badly. We detect the viewport and conditionally apply parallax. */
   261→   261→  const [isDesktop, setIsDesktop] = useState(false);
   262→   262→  useEffect(() => {
   263→   263→    const check = () => setIsDesktop(window.innerWidth >= 768);
   264→   264→    check();
   265→   265→    window.addEventListener('resize', check);
   266→   266→    return () => window.removeEventListener('resize', check);
   267→   267→  }, []);
   268→   268→  const { scrollY } = useScroll();
   269→   269→  const heroTextY = useTransform(scrollY, [0, 700], isDesktop ? [0, -210] : [0, 0]);
   270→   270→  const heroVizY = useTransform(scrollY, [0, 700], isDesktop ? [0, -350] : [0, 0]);
   271→   271→  const bgGridY = useTransform(scrollY, [0, 4000], isDesktop ? [0, 600] : [0, 0]);
   272→   272→  const parallaxWillChange = isDesktop ? 'transform' as const : undefined;
   273→   273→
   274→   274→  /* Close mobile drawer on Escape */
   275→   275→  useEffect(() => {
   276→   276→    if (!mobileNavOpen) return;
   277→   277→    const handler = (e: KeyboardEvent) => {
   278→   278→      if (e.key === 'Escape') setMobileNavOpen(false);
   279→   279→    };
   280→   280→    window.addEventListener('keydown', handler);
   281→   281→    return () => window.removeEventListener('keydown', handler);
   282→   282→  }, [mobileNavOpen]);
   283→   283→
   284→   284→  /* GSAP-animated Fourier series epicycles — infinite rotation */
   285→   285→  useEffect(() => {
   286→   286→    if (typeof window === 'undefined') return;
   287→   287→    const prefersReduced =
   288→   288→      typeof window !== 'undefined' &&
   289→   289→      window.matchMedia('(prefers-reduced-motion: reduce)').matches;
   290→   290→
   291→   291→    const tweens: gsap.core.Tween[] = [];
   292→   292→    const timelines: gsap.core.Timeline[] = [];
   293→   293→
   294→   294→    // ─────────────────────────────────────────────────────────────────────────
   295→   295→    // Fourier series for a square wave:
   296→   296→    //   f(t) = (4/π) · Σ_{k=1..7} sin((2k-1)·ω·t) / (2k-1)
   297→   297→    //
   298→   298→    // 7 epicycles chained together. Each circle's center is the previous
   299→   299→    // circle's tip; the tip of the last circle traces a complex curve whose
   300→   300→    // y-coordinate equals the Fourier sum (≈ a square wave).
   301→   301→    //
   302→   302→    // The waveform on the right half of the SVG shows the last 6 seconds of
   303→   303→    // the tip's y-position, recomputed each frame, scrolling leftward as t
   304→   304→    // advances.
   305→   305→    // ─────────────────────────────────────────────────────────────────────────
   306→   306→
   307→   307→    const FOURIER_SCALE = 36;       // px per unit of amplitude (large enough to see)
   308→   308→    const FOURIER_OMEGA_BASE = 1.4; // rad/s for the base frequency (k=1)
   309→   309→    const WAVE_SAMPLES = 120;       // polyline resolution
   310→   310→    const WAVE_WINDOW = 6;         // seconds of waveform visible on the right
   311→   311→    const WAVE_X_START = 120;
   312→   312→    const WAVE_X_END = 240;
   313→   313→    const WAVE_X_RANGE = WAVE_X_END - WAVE_X_START;
   314→   314→    const EPICYCLE_CENTER_X = 60;
   315→   315→    const EPICYCLE_CENTER_Y = 80;
   316→   316→
   317→   317→    const omegas = HARMONIC_KS.map((k) => k * FOURIER_OMEGA_BASE);
   318→   318→
   319→   319→    function fourierSum(s: number): number {
   320→   320→      let sum = 0;
   321→   321→      for (let i = 0; i < HARMONIC_KS.length; i++) {
   322→   322→        sum += HARMONIC_AMPS[i] * Math.sin(omegas[i] * s);
   323→   323→      }
   324→   324→      return sum;
   325→   325→    }
   326→   326→
   327→   327→    let t = 0;
   328→   328→    let lastFrameTime = performance.now();
   329→   329→
   330→   330→    function renderFrame(currentTime: number) {
   331→   331→      // Clamp dt to 50ms to avoid huge jumps after tab-switch.
   332→   332→      const dt = Math.min((currentTime - lastFrameTime) / 1000, 0.05);
   333→   333→      lastFrameTime = currentTime;
   334→   334→      t += dt;
   335→   335→
   336→   336→      // Walk the epicycle chain — each tip becomes the next circle's center.
   337→   337→      let prevX = EPICYCLE_CENTER_X;
   338→   338→      let prevY = EPICYCLE_CENTER_Y;
   339→   339→      let sumValue = 0;
   340→   340→
   341→   341→      for (let i = 0; i < HARMONIC_KS.length; i++) {
   342→   342→        const angle = omegas[i] * t;
   343→   343→        const dx = HARMONIC_AMPS[i] * FOURIER_SCALE * Math.cos(angle);
   344→   344→        // Negate sin for SVG y-down convention (positive sum → tip moves up).
   345→   345→        const dy = -HARMONIC_AMPS[i] * FOURIER_SCALE * Math.sin(angle);
   346→   346→        const tipX = prevX + dx;
   347→   347→        const tipY = prevY + dy;
   348→   348→
   349→   349→        const circleEl = circleRefs.current[i];
   350→   350→        if (circleEl) {
   351→   351→          circleEl.setAttribute('cx', prevX.toFixed(2));
   352→   352→          circleEl.setAttribute('cy', prevY.toFixed(2));
   353→   353→        }
   354→   354→        const lineEl = lineRefs.current[i];
   355→   355→        if (lineEl) {
   356→   356→          lineEl.setAttribute('x1', prevX.toFixed(2));
   357→   357→          lineEl.setAttribute('y1', prevY.toFixed(2));
   358→   358→          lineEl.setAttribute('x2', tipX.toFixed(2));
   359→   359→          lineEl.setAttribute('y2', tipY.toFixed(2));
   360→   360→        }
   361→   361→
   362→   362→        prevX = tipX;
   363→   363→        prevY = tipY;
   364→   364→        sumValue += HARMONIC_AMPS[i] * Math.sin(omegas[i] * t);
   365→   365→      }
   366→   366→
   367→   367→      // Tip marker
   368→   368→      if (tipRef.current) {
   369→   369→        tipRef.current.setAttribute('cx', prevX.toFixed(2));
   370→   370→        tipRef.current.setAttribute('cy', prevY.toFixed(2));
   371→   371→      }
   372→   372→
   373→   373→      // Waveform polyline — recompute each frame from past time samples.
   374→   374→      // x=240 is now (t), x=120 is WAVE_WINDOW seconds ago. As t advances,
   375→   375→      // the entire waveform naturally scrolls leftward.
   376→   376→      if (waveformRef.current) {
   377→   377→        let points = '';
   378→   378→        for (let i = 0; i < WAVE_SAMPLES; i++) {
   379→   379→          const x = WAVE_X_START + (i / (WAVE_SAMPLES - 1)) * WAVE_X_RANGE;
   380→   380→          const timeAtX = t - (1 - i / (WAVE_SAMPLES - 1)) * WAVE_WINDOW;
   381→   381→          const y = EPICYCLE_CENTER_Y - fourierSum(timeAtX) * FOURIER_SCALE;
   382→   382→          points += `${x.toFixed(1)},${y.toFixed(1)} `;
   383→   383→        }
   384→   384→        waveformRef.current.setAttribute('points', points.trim());
   385→   385→      }
   386→   386→
   387→   387→      // Trail line — horizontal connector from the tip to the rightmost
   388→   388→      // (newest) waveform sample.
   389→   389→      if (trailLineRef.current) {
   390→   390→        trailLineRef.current.setAttribute('x1', prevX.toFixed(2));
   391→   391→        trailLineRef.current.setAttribute('y1', prevY.toFixed(2));
   392→   392→        trailLineRef.current.setAttribute('x2', String(WAVE_X_END));
   393→   393→        trailLineRef.current.setAttribute('y2', prevY.toFixed(2));
   394→   394→      }
   395→   395→
   396→   396→      // Live readout
   397→   397→      if (tLabelRef.current) tLabelRef.current.textContent = (t % 10).toFixed(2);
   398→   398→      if (sumLabelRef.current) sumLabelRef.current.textContent = sumValue.toFixed(3);
   399→   399→    }
   400→   400→
   401→   401→    // ── Initial render ──
   402→   402→    renderFrame(performance.now());
   403→   403→
   404→   404→    // ── Entrance timeline (one-shot) ──
   405→   405→    const tl = gsap.timeline({ defaults: { ease: 'power3.out' } });
   406→   406→    timelines.push(tl);
   407→   407→    if (sceneRef.current) {
   408→   408→      tl.fromTo(sceneRef.current, { scale: 0.92, opacity: 0, y: 24 }, { scale: 1, opacity: 1, y: 0, duration: 0.7 });
