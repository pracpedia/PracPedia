   110→const maskDatabaseUrl = (url: string | undefined): string => {
   111→  if (!url) return '(not set)';
   112→  // For file:./prisma/dev.db-style SQLite URLs, show just the path tail
   113→  if (url.startsWith('file:')) {
   114→    const parts = url.replace('file:', '').split('/');
   115→    return `file:./${parts[parts.length - 1]}`;
   116→  }
   117→  // For postgres/etc URLs, mask credentials
   118→  try {
   119→    if (url.includes('://')) {
   120→      const u = new URL(url);
   121→      const host = u.hostname || 'db-host';
   122→      const path = u.pathname ? `/${u.pathname}` : '';
   123→      return `${u.protocol}//••••@${host}${path}`;
   124→    }
   125→  } catch {
   126→    /* fall through */
   127→  }
   128→  // Fallback: show last 2 segments
   129→  const parts = url.split(/[\/\\]/).filter(Boolean);
   130→  return parts.length > 2 ? `…/${parts.slice(-2).join('/')}` : url;
   131→};
