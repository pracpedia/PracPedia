   504→  // Synthesize a beautiful, clean, luxury notification chime to alert students in a modern academy
   505→  const triggerHorrorScreamerSound = () => {
   506→    try {
   507→      const AudioContextClass = window.AudioContext || (window as any).webkitAudioContext;
   508→      if (!AudioContextClass) return;
   509→      const ctx = new AudioContextClass();
   510→
   511→      const playChime = (freq: number, delay: number, dur: number) => {
   512→        const osc = ctx.createOscillator();
   513→        const gain = ctx.createGain();
   514→        osc.type = "sine";
   515→        osc.frequency.setValueAtTime(freq, ctx.currentTime + delay);
   516→        gain.gain.setValueAtTime(0, ctx.currentTime + delay);
   517→        gain.gain.linearRampToValueAtTime(0.25, ctx.currentTime + delay + 0.04);
   518→        gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + delay + dur);
   519→        osc.connect(gain);
   520→        gain.connect(ctx.destination);
   521→        osc.start(ctx.currentTime + delay);
   522→        osc.stop(ctx.currentTime + delay + dur);
   523→      };
   524→
   525→      // Play a premium major-seventh ambient chord progression (C Major 7)
   526→      playChime(523.25, 0.0, 1.2); // C5

   528→      playChime(783.99, 0.16, 1.6); // G5
