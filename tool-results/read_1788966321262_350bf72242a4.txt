   527→  // Relative view distance helper for 'Recently Viewed' label
   528→  const getRecentlyViewedMsg = (folderId: string) => {
   529→    const timestamp = recentlyViewed[folderId];
   530→    if (!timestamp) return null;
   531→    const diff = Date.now() - timestamp;
   532→    if (diff < 60000) return 'Just now';
   533→    const mins = Math.floor(diff / 60000);
   534→    if (mins < 60) return `${mins}m ago`;
   535→    const hrs = Math.floor(diff / 3600000);
   536→    if (hrs < 24) return `${hrs}h ago`;
   537→    const days = Math.floor(hrs / 24);
   538→    return `${days}d ago`;
   539→  };
