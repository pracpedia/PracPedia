   502→  // Synthesize a beautiful, clean, luxury notification chime to alert students in a modern academy
   503→  const triggerHorrorScreamerSound = () => {
   504→    try {
   505→      const AudioContextClass = window.AudioContext || (window as any).webkitAudioContext;
   506→      if (!AudioContextClass) return;
   507→      const ctx = new AudioContextClass();
   508→
   509→      const playChime = (freq: number, delay: number, dur: number) => {
   510→        const osc = ctx.createOscillator();
   511→        const gain = ctx.createGain();
   512→        osc.type = "sine";
   513→        osc.frequency.setValueAtTime(freq, ctx.currentTime + delay);
   514→        gain.gain.setValueAtTime(0, ctx.currentTime + delay);
   515→        gain.gain.linearRampToValueAtTime(0.25, ctx.currentTime + delay + 0.04);
   516→        gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + delay + dur);
   517→        osc.connect(gain);
   518→        gain.connect(ctx.destination);
   519→        osc.start(ctx.currentTime + delay);
   520→        osc.stop(ctx.currentTime + delay + dur);
   521→      };
   522→
   523→      // Play a premium major-seventh ambient chord progression (C Major 7)
   524→      playChime(523.25, 0.0, 1.2); // C5
   525→      playChime(659.25, 0.08, 1.4); // E5
   526→      playChime(783.99, 0.16, 1.6); // G5
   527→      playChime(987.77, 0.24, 2.0); // B5
   528→    } catch (e) {
   529→      // Silent — browsers block AudioContext until user gesture (autoplay policy).
   530→      // This is expected behavior, not a bug, so we don't pollute the console.
