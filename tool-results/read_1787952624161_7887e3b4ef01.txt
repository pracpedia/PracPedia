   172→/* ---------------- AnimatedStat ---------------- */
   173→
   174→const AnimatedStat: React.FC<{ value: number; className?: string }> = ({ value, className = '' }) => {
   175→  const ref = useRef<HTMLSpanElement>(null);
   176→  const inView = useInView(ref, { once: true, margin: '-40px' });
   177→  const [display, setDisplay] = useState(0);
   178→
   179→  useEffect(() => {
   180→    if (!inView) return;
   181→    const prefersReducedMotion =
   182→      typeof window !== 'undefined' && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
   183→    if (prefersReducedMotion) {
   184→      setDisplay(value);
   185→      return;
   186→    }
   187→    let raf = 0;
   188→    const start = performance.now();
   189→    const duration = 1400;
   190→    const tick = (now: number) => {
   191→      const t = Math.min((now - start) / duration, 1);
   192→      const eased = 1 - Math.pow(1 - t, 3);
   193→      setDisplay(Math.round(eased * value));
   194→      if (t < 1) raf = requestAnimationFrame(tick);
   195→    };
   196→    raf = requestAnimationFrame(tick);
   197→    return () => cancelAnimationFrame(raf);
   198→  }, [inView, value]);
   199→
   200→  return <span ref={ref} className={className}>{display}</span>;
   201→};

   268→/** Particle field — renders N particles distributed across the parent. */
   269→const ParticleField: React.FC<{ count?: number; className?: string }> = ({
   270→  count = 18,
   271→  className = '',
   272→}) => {
   273→  // Deterministic pseudo-random so the field is stable between renders (no
   274→  // hydration mismatch with SSR).
   275→  const particles = useMemo(() => {
   276→    const seed = 1337;
   277→    let s = seed;
   278→    const rand = () => {
   279→      // Mulberry32 PRNG — fast, no deps
   280→      s |= 0; s = (s + 0x6d2b79f5) | 0;
   281→      let t = Math.imul(s ^ (s >>> 15), 1 | s);
   282→      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   283→      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   284→    };
   285→    const colors = ['#22d3ee', '#a78bfa', '#fbbf24', '#34d399', '#f472b6'];
   286→    return Array.from({ length: count }).map((_, i) => ({
   287→      id: i,
   288→      x: `${Math.floor(rand() * 100)}%`,
   289→      size: 2 + Math.floor(rand() * 4),
   290→      color: colors[Math.floor(rand() * colors.length)],
   291→      duration: 8 + rand() * 8,
   292→      delay: rand() * 8,
   293→    }));
   294→  }, [count]);
   295→  return (
   296→    <div aria-hidden className={`pointer-events-none absolute inset-0 overflow-hidden ${className}`}>
   297→      {particles.map((p) => (
   298→        <Particle key={p.id} x={p.x} size={p.size} color={p.color} duration={p.duration} delay={p.delay} />
   299→      ))}
   300→    </div>
   301→  );
   302→};
