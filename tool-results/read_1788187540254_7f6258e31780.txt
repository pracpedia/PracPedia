     1→'use client';
     2→
     3→import React, { useState, useEffect, useRef } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import { useAuth } from '@/contexts/AuthContext';
     6→import { safeLocalStorage } from '@/lib/storage';
     7→import { FormattedMarkdown } from './FormattedMarkdown';
     8→import {
     9→  ChevronLeft,
    10→  ChevronRight,
    11→  X,
    12→  ZoomIn,
    13→  ZoomOut,
    14→  Sparkles,
    15→  Cpu,
    16→  BookOpen,
    17→  MessageSquare,
    18→  Send,
    19→  HelpCircle,
    20→  FlaskConical,
    21→  RefreshCw,
    22→  CheckCircle2,
    23→  AlertTriangle,
    24→} from 'lucide-react';
    25→
    26→interface LightboxProps {
    27→  images: { url: string; title?: string }[];
    28→  initialIndex: number;
    29→  onClose: () => void;
    30→}
    31→
    32→export const Lightbox: React.FC<LightboxProps> = ({ images, initialIndex, onClose }) => {
    33→  const { apiFetch, user, setUser } = useAuth();
    34→
    35→  // Carousel states
    36→  const [index, setIndex] = useState(initialIndex);
    37→  const [zoom, setZoom] = useState(false);
    38→
    39→  // HSC Science AI Helper States
    40→  const [showAiAssistant, setShowAiAssistant] = useState(false);
    41→  const [isAnalyzing, setIsAnalyzing] = useState(false);
    42→  const [analysisResult, setAnalysisResult] = useState<string | null>(null);
    43→  const [errorMessage, setErrorMessage] = useState<string | null>(null);
    44→
    45→  // Custom Chat States
    46→  const [activeTab, setActiveTab] = useState<'guide' | 'chat'>('guide');
    47→  const [chatLogs, setChatLogs] = useState<Array<{ sender: 'student' | 'ai'; text: string }>>([]);
    48→  const [customQuestion, setCustomQuestion] = useState('');
    49→  const [isAnswering, setIsAnswering] = useState(false);
    50→  const [aiLanguage, setAiLanguage] = useState<'en' | 'bn_book'>(() => {
    51→    if (typeof window === 'undefined') return 'en';
    52→    return (safeLocalStorage.getItem('ai_language') as 'en' | 'bn_book') || 'en';
    53→  });
    54→
    55→  const handleLanguageChange = (lang: 'en' | 'bn_book') => {
    56→    setAiLanguage(lang);
    57→    safeLocalStorage.setItem('ai_language', lang);
    58→  };
    59→
    60→  const [rechargingCredits, setRechargingCredits] = useState(false);
    61→
    62→  const handleRechargeTrial = async () => {
    63→    setRechargingCredits(true);
    64→    try {
    65→      const res = await apiFetch('/api/users/recharge-trial', {
    66→        method: 'POST',
    67→      });
    68→      if (res.ok) {
    69→        const data = await res.json();
    70→        if (user && setUser) {
    71→          setUser({ ...user, aiCredits: data.aiCredits });
    72→        }
    73→        setErrorMessage(null);
    74→      } else {
    75→        const errData = await res.json();
    76→        alert(errData.error || "Failed to recharge trial credits.");
    77→      }
    78→    } catch (e) {
    79→      alert("Error contacting the credit recharge service.");
    80→    } finally {
    81→      setRechargingCredits(false);
    82→    }
    83→  };
    84→
    85→  // Automatic scroll container reference
    86→  const chatScrollRef = useRef<HTMLDivElement>(null);
    87→
    88→  useEffect(() => {
    89→    const container = chatScrollRef.current;
    90→    if (!container) return;
    91→
    92→    // Pin scroll position directly to bottom to prevent outer iframe scroll jitter
    93→    container.scrollTop = container.scrollHeight;
    94→
    95→    // Use ResizeObserver to auto-scroll when math blocks/LaTex images render asynchronously
    96→    const resizeObserver = new ResizeObserver(() => {
    97→      container.scrollTop = container.scrollHeight;
    98→    });
    99→
   100→    resizeObserver.observe(container);
   101→
   102→    // Fallback timer to double-check layout adjustments
   103→    const timer = setTimeout(() => {
   104→      container.scrollTop = container.scrollHeight;
   105→    }, 150);
   106→
   107→    return () => {
   108→      resizeObserver.disconnect();
   109→      clearTimeout(timer);
   110→    };
   111→  }, [chatLogs, isAnswering]);
   112→
   113→  // Reset state when swiping notebook sheets
   114→  const handleIndexChange = (newIdx: number) => {
   115→    setIndex(newIdx);
   116→    setZoom(false);
   117→    setAnalysisResult(null);
   118→    setChatLogs([]);
   119→    setErrorMessage(null);
   120→    setActiveTab('guide');
   121→  };
   122→
   123→  const handlePrev = (e?: React.MouseEvent) => {
   124→    e?.stopPropagation();
   125→    const newIdx = index > 0 ? index - 1 : images.length - 1;
   126→    handleIndexChange(newIdx);
   127→  };
   128→
   129→  const handleNext = (e?: React.MouseEvent) => {
   130→    e?.stopPropagation();
   131→    const newIdx = index < images.length - 1 ? index + 1 : 0;
   132→    handleIndexChange(newIdx);
   133→  };
   134→
   135→  const currentImage = images[index];
   136→
   137→  // Call backend parser
   138→  const handleAnalyzePage = async () => {
   139→    if (!currentImage?.url) return;
   140→    try {
   141→      setIsAnalyzing(true);
   142→      setErrorMessage(null);
   143→
   144→      const res = await apiFetch('/api/scan/analyze-page', {
   145→        method: 'POST',
   146→        headers: { 'Content-Type': 'application/json' },
   147→        body: JSON.stringify({ imageUrl: currentImage.url, language: aiLanguage })
   148→      });
   149→
   150→      if (!res.ok) {
   151→        throw new Error("Unable to read image scan. Check your internet connection.");
   152→      }
   153→
   154→      const data = await res.json();
   155→      if (typeof data.aiCredits === 'number' && user) {
   156→        setUser({ ...user, aiCredits: data.aiCredits });
   157→      }
   158→      setAnalysisResult(data.analysis);
   159→
   160→      setChatLogs([
   161→        {
   162→          sender: 'ai',
   163→          text: aiLanguage === 'bn_book'
   164→            ? `👋 আসসালামু আলাইকুম! আমি আপনার ব্যবহারিক নোটবুকের পৃষ্ঠাটি স্ক্যান করেছি। আমি এই অংশটির মূল তাত্ত্বিক ধারণা, সমীকরণ এবং রিডিং বুঝতে সাহায্য করতে প্রস্তুত। আপনি কি নিয়ে আলোচনা করতে চান?`
   165→            : `👋 Salam! I have scanned this practical notebook sheet. I am ready to explain the underlying theories, balance equations, or help you understand this table. What would you like to discuss?`
   166→        }
   167→      ]);
   168→    } catch (err: any) {
   169→      setErrorMessage(err.message || "Failed to establish connection with AI processing node.");
   170→    } finally {
   171→      setIsAnalyzing(false);
   172→    }
   173→  };
   174→
   175→  // Submit student chat questions
   176→  const handleAskQuestion = async (presetText?: string) => {
   177→    const textToSend = (presetText || customQuestion).trim();
   178→    if (!textToSend || !currentImage?.url) return;
   179→
   180→    setChatLogs((prev) => [...prev, { sender: 'student', text: textToSend }]);
   181→    if (!presetText) setCustomQuestion('');
   182→
   183→    try {
   184→      setIsAnswering(true);
   185→      setErrorMessage(null);
   186→
   187→      const res = await apiFetch('/api/scan/analyze-page', {
   188→        method: 'POST',
   189→        headers: { 'Content-Type': 'application/json' },
   190→        body: JSON.stringify({
   191→          imageUrl: currentImage.url,
   192→          question: textToSend,
   193→          language: aiLanguage
   194→        })
   195→      });
   196→
   197→      if (!res.ok) {
   198→        throw new Error("Datalink failed to gather academic assessment.");
   199→      }
   200→
   201→      const data = await res.json();
   202→      if (typeof data.aiCredits === 'number' && user) {
   203→        setUser({ ...user, aiCredits: data.aiCredits });
   204→      }
   205→      setChatLogs((prev) => [...prev, { sender: 'ai', text: data.analysis }]);
   206→    } catch (err: any) {
   207→      setChatLogs((prev) => [...prev, { sender: 'ai', text: `⚠️ System Error: ${err.message || "Could not retrieve prompt reply."}` }]);
   208→    } finally {
   209→      setIsAnswering(false);
   210→    }
   211→  };
   212→
   213→  // Custom Markdown parser for rich formatted HSC typography
   214→  const renderHscMarkdownList = (text: string) => {
   215→    if (!text) return null;
   216→    const lines = text.split('\n');
   217→
   218→    return (
   219→      <div className="space-y-3.5 font-sans text-xs sm:text-[13px] leading-relaxed text-slate-300">
   220→        {lines.map((line, idx) => {
   221→          const trimmed = line.trim();
   222→          if (!trimmed) return <div key={idx} className="h-1" />;
   223→
   224→          // Headers formatting
   225→          if (trimmed.startsWith('# ')) {
   226→            return (
   227→              <h3 key={idx} className="text-sm sm:text-base font-black text-white mt-5 mb-2.5 pb-1 border-b border-white/5 flex items-center gap-2 break-words">
   228→                <span className="w-1 h-3.5 bg-gradient-to-b from-cyan-400 to-indigo-500 rounded-full shrink-0" />
   229→                {trimmed.substring(2)}
   230→              </h3>
   231→            );
   232→          }
   233→          if (trimmed.startsWith('## ') || trimmed.startsWith('### ')) {
   234→            const titleText = trimmed.startsWith('## ') ? trimmed.substring(3) : trimmed.substring(4);
   235→            return (
   236→              <h4 key={idx} className="text-xs sm:text-[12px] font-bold text-cyan-400 mt-4 mb-2 tracking-wide uppercase font-mono flex items-center gap-1.5 break-words">
   237→                <Sparkles className="w-3 h-3 text-cyan-400 animate-pulse shrink-0" />
   238→                {titleText}
   239→              </h4>
   240→            );
   241→          }
   242→
   243→          let content = trimmed;
   244→          const isBullet = trimmed.startsWith('- ') || trimmed.startsWith('* ');
   245→          if (isBullet) {
   246→            content = trimmed.substring(2);
   247→          }
   248→
   249→          // Simple code block display helper (formulas, fractions, etc.)
   250→          const isFormula = content.includes('$$') || content.includes('\\frac') || content.includes('\\pi');
   251→
   252→          // Bold parsing (**bold**)
   253→          const parts = content.split('**');
   254→          const renderedLine = parts.map((part, pIdx) => {
   255→            if (pIdx % 2 === 1) {
   256→              return <strong key={pIdx} className="text-amber-300 font-extrabold">{part}</strong>;
   257→            }
   258→            return <React.Fragment key={pIdx}>{part}</React.Fragment>;
   259→          });
   260→
   261→          if (isBullet) {
   262→            return (
   263→              <div key={idx} className="flex items-start gap-2 ml-1 mt-1 font-sans">
   264→                <span className="w-1.5 h-1.5 rounded-full bg-cyan-500 mt-2 shrink-0 animate-pulse" />
   265→                <div className="flex-1 text-slate-300 font-medium break-words">{renderedLine}</div>
   266→              </div>
   267→            );
   268→          }
   269→
   270→          if (isFormula) {
   271→            return (
   272→              <div key={idx} className="my-2.5 p-3 rounded-xl bg-slate-950/80 border border-cyan-900/30 text-center font-mono text-cyan-300 select-all overflow-x-auto text-[11px] leading-relaxed shadow-inner break-words">
   273→                {content.replace(/\$\$/g, '').trim()}
   274→              </div>
   275→            );
   276→          }
   277→
   278→          return <p key={idx} className="text-slate-350 font-normal leading-relaxed break-words">{renderedLine}</p>;
   279→        })}
   280→      </div>
   281→    );
   282→  };
   283→
   284→  // Reference recharge function so it remains part of the component surface (used by parent later)
   285→  void rechargingCredits;
   286→  void handleRechargeTrial;
   287→
   288→  return (
   289→    <AnimatePresence>
   290→      <motion.div
   291→        initial={{ opacity: 0 }}
   292→        animate={{ opacity: 1 }}
   293→        exit={{ opacity: 0 }}
   294→        onClick={(e) => {
   295→          if (e.target === e.currentTarget) onClose();
   296→        }}
   297→        className="fixed inset-0 z-50 flex flex-col lg:flex-row bg-[#030712]/98 backdrop-blur-xl select-none overflow-y-auto lg:overflow-hidden text-slate-100"
   298→      >
   299→
   300→        {/* ========================================================
   301→            LEFT PORTION: RESPONSIVE CAROUSEL & IMAGE VIEWER
   302→            ======================================================== */}
   303→        <div className="flex-1 flex flex-col justify-between p-3 sm:p-5 h-[48vh] sm:h-[50vh] lg:h-full relative overflow-hidden select-none border-b lg:border-b-0 lg:border-r border-white/5 bg-[#090d16]/30 min-h-0">
   304→
   305→          {/* Top Panel Header Actions */}
   306→          <div className="flex items-center justify-between w-full text-white/90 p-1.5 z-10 select-none shrink-0 gap-2">
   307→            <div className="flex flex-col text-left min-w-0">
   308→              <span className="text-[8px] font-mono tracking-widest text-cyan-400 font-black uppercase flex items-center gap-1">
   309→                <FlaskConical className="w-3 h-3 text-cyan-400 shrink-0" />
   310→                <span className="truncate">HSC NOTEBOOK SCANNER</span>
   311→              </span>
   312→              <span className="text-xs sm:text-sm font-black text-slate-150 truncate max-w-[140px] sm:max-w-md mt-0.5" title={currentImage?.title}>
   313→                {currentImage?.title || `Notebook Page ${index + 1}`}
   314→              </span>
   315→            </div>
   316→
   317→            <div className="flex items-center gap-1.5 sm:gap-2 shrink-0">
   318→              {/* HSC AI Toggle */}
   319→              <button
   320→                onClick={(e) => {
   321→                  e.stopPropagation();
   322→                  setShowAiAssistant(!showAiAssistant);
   323→                }}
   324→                className={`flex items-center gap-1.5 px-2 sm:px-3 py-2 min-h-[44px] rounded-xl text-[10px] font-black font-mono uppercase tracking-wide border-2 transition-all cursor-pointer ${
   325→                  showAiAssistant
   326→                    ? 'bg-cyan-500/10 border-cyan-400 text-cyan-300 animate-pulse shadow-[0_0_15px_rgba(34,211,238,0.15)]'
   327→                    : 'bg-white/5 border-white/10 hover:border-white/30 text-slate-300 hover:text-white'
   328→                }`}
   329→              >
   330→                <Cpu className="w-3.5 h-3.5 text-cyan-400 shrink-0" />
   331→                <span className="hidden sm:inline">AI Teacher</span>
   332→                <span className="sm:hidden">AI</span>
   333→              </button>
   334→
   335→              <span className="px-2.5 py-1.5 rounded-xl bg-white/5 border border-white/5 text-[9.5px] font-mono font-bold text-slate-400 shrink-0">
   336→                {index + 1} / {images.length}
   337→              </span>
   338→
   339→              <button
   340→                onClick={(e) => {
   341→                  e.stopPropagation();
   342→                  setZoom(!zoom);
   343→                }}
   344→                className="p-2 min-h-[44px] min-w-[44px] flex items-center justify-center rounded-xl bg-white/5 border border-white/10 text-slate-300 hover:text-white transition-colors cursor-pointer hidden md:flex"
   345→                title={zoom ? "Zoom Out" : "Zoom In"}
   346→              >
   347→                {zoom ? <ZoomOut className="w-4 h-4" /> : <ZoomIn className="w-4 h-4" />}
   348→              </button>
   349→
   350→              <button
   351→                onClick={onClose}
   352→                className="p-2 min-h-[44px] min-w-[44px] flex items-center justify-center rounded-xl bg-rose-500/10 border border-rose-500/20 hover:bg-rose-600 hover:border-transparent text-rose-300 hover:text-white transition-all cursor-pointer shadow-sm"
   353→                title="Exit viewer"
   354→              >
   355→                <X className="w-4 h-4" />
   356→              </button>
   357→            </div>
   358→          </div>
   359→
   360→          {/* Center Image viewport (extremely fit, non-distorting on any phone) */}
   361→          <div className="relative flex-1 flex items-center justify-center min-h-0 w-full select-none py-2 shrink-0">
   362→            {images.length > 1 && (
   363→              <button
   364→                onClick={handlePrev}
   365→                className="absolute left-1 sm:left-3 z-10 p-2 sm:p-3 min-h-[44px] min-w-[44px] flex items-center justify-center rounded-xl bg-slate-900/40 border border-white/5 hover:bg-white/10 text-white transition-all cursor-pointer active:scale-90"
   366→                aria-label="Previous image"
   367→              >
   368→                <ChevronLeft className="w-5 h-5" />
   369→              </button>
   370→            )}
   371→
   372→            <div
   373→              onClick={(e) => e.stopPropagation()}
   374→              className="w-full h-full flex items-center justify-center p-2 relative min-h-0"
   375→            >
   376→              <motion.img
   377→                key={index}
   378→                initial={{ scale: 0.97, opacity: 0 }}
   379→                animate={{
   380→                  scale: zoom ? 1.2 : 1,
   381→                  opacity: 1,
   382→                  cursor: zoom ? 'zoom-out' : 'zoom-in'
   383→                }}
   384→                exit={{ scale: 0.97, opacity: 0 }}
   385→                transition={{ type: "spring", damping: 30, stiffness: 220 }}
   386→                src={currentImage?.url}
   387→                alt={currentImage?.title}
   388→                onClick={() => setZoom(!zoom)}
   389→                className="max-w-[95vw] max-h-[34vh] sm:max-h-[38vh] lg:max-h-[72vh] object-contain rounded-2xl border border-white/10 bg-slate-950/60 p-1.5 shadow-[0_15px_50px_rgba(0,0,0,0.5)]"
   390→              />
   391→            </div>
   392→
   393→            {images.length > 1 && (
   394→              <button
   395→                onClick={handleNext}
   396→                className="absolute right-1 sm:right-3 z-10 p-2 sm:p-3 min-h-[44px] min-w-[44px] flex items-center justify-center rounded-xl bg-slate-900/40 border border-white/5 hover:bg-white/10 text-white transition-all cursor-pointer active:scale-90"
   397→                aria-label="Next image"
   398→              >
   399→                <ChevronRight className="w-5 h-5" />
   400→              </button>
   401→            )}
   402→          </div>
   403→
   404→          {/* Bottom Thumbnails */}
   405→          <div className="w-full max-w-2xl mx-auto flex items-center justify-center gap-1.5 overflow-x-auto py-1 z-10 select-none shrink-0 pb-2">
   406→            {images.map((img, idx) => (
   407→              <button
   408→                key={idx}
   409→                onClick={(e) => {
   410→                  e.stopPropagation();
   411→                  handleIndexChange(idx);
   412→                }}
   413→                className={`relative h-8 w-11 rounded-lg overflow-hidden shrink-0 border-2 transition-all cursor-pointer ${
   414→                  idx === index ? 'border-cyan-400 scale-105 shadow-md shadow-cyan-500/20' : 'border-transparent opacity-40 hover:opacity-85'
   415→                }`}
   416→              >
   417→                <img src={img.url} className="w-full h-full object-cover" alt="" />
   418→              </button>
   419→            ))}
   420→          </div>
   421→        </div>
   422→
   423→
   424→        {/* ========================================================
   425→            RIGHT PORTION: MODERN, ROBUST HSC SCIENCE AI ASSISTANT
   426→            ======================================================== */}
   427→        <AnimatePresence>
   428→          {showAiAssistant ? (
   429→            <motion.div
   430→              initial={{ opacity: 0, x: typeof window !== 'undefined' && window.innerWidth < 1024 ? 0 : 360, y: typeof window !== 'undefined' && window.innerWidth < 1024 ? 120 : 0 }}
   431→              animate={{ opacity: 1, x: 0, y: 0 }}
   432→              exit={{ opacity: 0, x: typeof window !== 'undefined' && window.innerWidth < 1024 ? 0 : 360, y: typeof window !== 'undefined' && window.innerWidth < 1024 ? 120 : 0 }}
   433→              transition={{ type: "spring", damping: 28, stiffness: 200 }}
   434→              className="w-full lg:w-[480px] xl:w-[520px] bg-[#070b13] border-t lg:border-t-0 lg:border-l border-white/5 flex flex-col shrink-0 text-left h-[52vh] lg:h-full relative overflow-hidden z-20 shadow-[0_-15px_40px_rgba(0,0,0,0.6)] select-text min-h-0"
   435→              onClick={(e) => e.stopPropagation()}
   436→            >
   437→
   438→              {/* Dynamic Header */}
   439→              <div className="p-3 sm:p-4 bg-[#0a101b] border-b border-white/5 flex items-center justify-between shrink-0 select-none gap-2">
   440→                <div className="flex items-center gap-2 sm:gap-3 min-w-0">
   441→                  <div className="relative shrink-0">
   442→                    <div className="absolute -inset-1 rounded-xl bg-cyan-500/30 blur animate-pulse" />
   443→                    <div className="relative p-2 rounded-xl bg-cyan-950/60 border border-cyan-500/35 text-cyan-300">
   444→                      <Sparkles className="w-4 h-4" />
   445→                    </div>
   446→                  </div>
   447→                  <div className="min-w-0">
   448→                    <h4 className="text-xs sm:text-sm font-black text-white tracking-wide truncate">Bangladesh HSC Practical Guide</h4>
   449→                    <span className="text-[9px] font-mono text-cyan-400 font-extrabold uppercase tracking-widest block mt-0.5 truncate">MULTIMODAL LESSON GENERATOR</span>
   450→                  </div>
   451→                </div>
   452→
   453→                <div className="flex items-center gap-1 bg-slate-950 border border-white/10 p-0.5 rounded-lg select-none shrink-0">
   454→                  <button
   455→                    onClick={() => handleLanguageChange('en')}
   456→                    className={`px-2 py-1 min-h-[28px] rounded text-[9px] font-black uppercase tracking-wider transition-all cursor-pointer ${aiLanguage === 'en' ? 'bg-gradient-to-r from-cyan-500 to-indigo-600 text-slate-950 font-black shadow-md' : 'text-slate-400 hover:text-slate-200'}`}
   457→                  >
   458→                    English
   459→                  </button>
   460→                  <button
   461→                    onClick={() => handleLanguageChange('bn_book')}
   462→                    className={`px-2 py-1 min-h-[28px] rounded text-[9px] font-black uppercase tracking-wider transition-all cursor-pointer ${aiLanguage === 'bn_book' ? 'bg-gradient-to-r from-cyan-500 to-indigo-600 text-slate-950 font-black shadow-md' : 'text-slate-400 hover:text-slate-200'}`}
   463→                  >
   464→                    বাংলা
   465→                  </button>
   466→                </div>
   467→              </div>
   468→
   469→              {/* Gemini API Key indicator if key provided */}
   470→              {user && (
